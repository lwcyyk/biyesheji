class CodecSpy:
    def __init__(self, store):
        self.store = store

    def decrypt(self, nonce, body, aad):
        store = self.store
        store.decode_calls += 1
        store.trace.append("decode")
        if store.failure == "decode" or nonce != b"N" * 12 or aad != b"auth-lifecycle/online-state/schema-1":
            raise RuntimeError(store.secret)
        if body != store.blob[12:]:
            raise ValueError(store.secret)
        return body[:-16]


class MemoryStore:
    def __init__(self, state, connection, secret):
        self.connection, self.secret = connection, secret
        self.lock = threading.RLock()
        self.operations, self.operation_limit = 0, None
        self.key_calls, self.decode_calls = 0, 0
        self.telemetry_attempts, self.telemetry_successes = 0, 0
        self.select_dispatches, self.forbidden = 0, 0
        self.failure, self.select_hook = None, None
        self.C_telemetry_ns = 0
        self.trace, self.telemetry, self.sql_templates = [], [], []
        connection.execute("CREATE TABLE authority_state(id INTEGER PRIMARY KEY,version INTEGER,ciphertext BLOB)")
        connection.execute("CREATE TABLE authority_events(version INTEGER,operation TEXT)")
        connection.execute("INSERT INTO authority_events VALUES(120,'fixture_event')")
        connection.execute("PRAGMA user_version=2")
        self.replace_fixture(state)

    def replace_fixture(self, state):
        self.state = copy.deepcopy(state)
        for raw in self.state["requests"].values():
            raw["digest"] = digest({"caller": raw["origin"]["caller"], "request": raw["request"]})
        self.blob = b"N" * 12 + canonical(self.state) + b"T" * 16
        self.connection.execute("DELETE FROM authority_state")
        self.connection.execute("INSERT INTO authority_state VALUES(1,120,?)", (self.blob,))

    def execute(self, statement, parameters=()):
        if statement != a.BOUNDED_SELECT:
            self.forbidden += 1
            raise AssertionError("SOURCE_QUERY_NOT_ALLOWLISTED")
        if self.operation_limit is not None and self.operations >= self.operation_limit:
            raise RuntimeError("DATABASE_OPERATION_BUDGET")
        self.operations += 1
        self.trace.append("operation_increment")
        telemetry_started = time.perf_counter_ns()
        try:
            self.telemetry_attempts += 1
            self.trace.append("telemetry_attempt")
            if self.failure == "telemetry":
                raise RuntimeError(self.secret)
            self.telemetry_successes += 1
            self.telemetry.append({"kind": "SELECT", "number": self.operations, "bound_value_bytes": 8})
            self.trace.append("telemetry_success")
        finally:
            self.C_telemetry_ns += time.perf_counter_ns() - telemetry_started
        if self.select_hook is not None:
            self.select_hook()
        self.select_dispatches += 1
        self.sql_templates.append(statement)
        self.trace.append("sqlite_select")
        return self.connection.execute(statement, parameters)

    def cipher(self):
        self.key_calls += 1
        self.trace.append("key")
        if self.failure == "key":
            raise RuntimeError(self.secret)
        return CodecSpy(self)

    def call(self, *args):
        self.forbidden += 1
        raise AssertionError("OLD_DISPATCH_FORBIDDEN")

    def world_hash(self):
        rows = [(row["id"], row["version"], bytes(row["ciphertext"]).hex())
                for row in self.connection.execute("SELECT id,version,ciphertext FROM authority_state ORDER BY id")]
        events = [tuple(row) for row in self.connection.execute("SELECT version,operation FROM authority_events ORDER BY version")]
        return digest({"rows": rows, "events": events})

    def counters(self):
        return {"operations": self.operations, "telemetry_attempts": self.telemetry_attempts,
                "telemetry_successes": self.telemetry_successes, "select_dispatches": self.select_dispatches,
                "key_calls": self.key_calls, "decode_calls": self.decode_calls, "forbidden": self.forbidden}

    def __repr__(self):
        return "<OwnedMemoryStore redacted>"


class Transport:
    def __init__(self, env):
        self.env, self.calls = env, 0

    def call(self, operation, payload):
        self.calls += 1
        if operation != h.RIGHT:
            raise ValueError("OPERATION_DENIED")
        return a.dispatch_authority_read(self.env.mount, self.env.peer, payload)


class ActivationCases(unittest.TestCase):
    def setUp(self):
        root = CONFIG["fixture_root"]
        self.case_id = self._testMethodName.split("_")[1]
        self.spec = next(x for x in CONFIG["plan"]["A1"]["new_tests"] if x["id"] == self.case_id)
        self.seed = json.loads((root / "STATE_SEED.json").read_text(encoding="utf-8-sig"))
        self.canaries = json.loads((root / "CANARIES.json").read_text(encoding="utf-8-sig"))["values"]
        self.checks, self.calls, self.environments, self.batches = [], [], [], []
        self.sequence = 0
        self.started = time.perf_counter_ns()

    def tearDown(self):
        DETAILS.append({"id": self.case_id, "expected": self.spec["expected"], "checks": self.checks,
                        "calls": self.calls, "batches": self.batches,
                        "session_final": [self.safe_session(e) for e in self.environments],
                        "elapsed_ns": time.perf_counter_ns() - self.started})
        for env in self.environments:
            env.issuer.retire("epoch:retired")

    def check(self, condition, label):
        self.checks.append({"label": label, "satisfied": bool(condition)})
        if not condition:
            raise AssertionError(label)

    def safe_session(self, env):
        return {"authorization_ref": env.authorization, "source_ref": env.store.source_ref,
                "C_telemetry_ns": env.store.C_telemetry_ns, "registration": env.mount.registration.counters(),
                "store": env.store.counters(), "trace": list(env.store.trace),
                "telemetry": copy.deepcopy(env.store.telemetry), "host_audit_rows": list(env.audit)}

    def environment(self, *, store=None, state=None, ceiling=a.CEILING, rid="rF", limits=None):
        self.sequence += 1
        if store is None:
            store = MemoryStore(self.seed["state"] if state is None else state,
                                CONFIG["connect"](), self.canaries["exception"])
            store.source_ref = "mem:" + self.case_id
        clock = SimpleNamespace(now=1000)
        issuer = a.ActivationIssuer("epoch:a1:" + str(self.sequence), clock=lambda: clock.now)
        audit = []
        authorization = "auth:" + self.case_id + "." + str(self.sequence)
        ticket = issuer.authorize_offline(authorization_id=authorization, scope=self.seed["scope_A"],
            grant_ref="gA", rid=rid, manager_identity="h2_reader", principal_ref="manager:a1", actor="user",
            transport_ref="rpc:authority", scope_ref="scope:a1", producer_ref="producer:a1", source_ref="authority:a1",
            expires_ns=1000000000, expected_store=store, ceiling=ceiling, audit_sink=audit.append,
            limits=limits or h.HostReadLimits())
        before = store.counters()
        mount = a.mount_authority_host(store, ticket)
        self.check(store.counters() == before, "REGISTRATION_ZERO_SQL_KEY_DECODE_AND_QUOTA")
        identities = {"h2_reader": self.canaries["rpc_token"], "controller": self.canaries["access_token"]}
        permissions = {"h2_reader": [h.RIGHT], "controller": [h.RIGHT, "grant"]}
        peer = a.authenticate_authority_peer(mount, "Bearer " + self.canaries["rpc_token"], identities, permissions)
        env = SimpleNamespace(store=store, clock=clock, issuer=issuer, ticket=ticket, mount=mount,
                              peer=peer, identities=identities, permissions=permissions,
                              authorization=authorization, audit=audit)
        self.environments.append(env)
        return env

    def verify_public(self, public, rendered):
        self.check(len(canonical(public)) <= 4096, "SUMMARY_WITHIN_4096")
        self.check(all(public[k] is False for k in ("math_ready", "current_native_eligible", "global_complete")), "NO_NATIVE_MATH_COMPLETE")
        self.check(public["authority_write_action"] == "NONE" and public["obligation_revision"] is None,
                   "NO_WRITE_OR_O_REVISION")
        self.check(set(BASE_GAPS) <= set(public["missing_fields"]), "UNKNOWN_GAPS_PRESERVED")
        self.check(all(value not in rendered for value in self.canaries.values()), "ALL_SECRET_CANARIES_ABSENT")
        self.check(set(public) == set(h.summary_schema()["properties"]), "CLOSED_ROOT_SHAPE")

    def invoke(self, env, *, payload=None, context=None, permit=None, direct=None, bridge_failure=False):
        world = env.store.world_hash()
        before, session_before = env.store.counters(), env.mount.registration.counters()
        stdout, stderr, logs = io.StringIO(), io.StringIO(), io.StringIO()
        handler = logging.StreamHandler(logs)
        logger = logging.getLogger()
        logger.addHandler(handler)
        private_stage = []
        def observe(*args, **kwargs):
            value = ORIGINAL_READ(*args, **kwargs)
            private_stage.append(value.to_dict())
            if bridge_failure:
                raise RuntimeError(self.canaries["exception"])
            return value
        started = time.perf_counter_ns()
        try:
            with patch.object(h, "read_dsat_authority_snapshot", observe), contextlib.redirect_stdout(stdout), contextlib.redirect_stderr(stderr):
                public = direct() if direct else a.dispatch_authority_read(env.mount,
                    env.peer if context is None else context, env.mount.refs(permit) if payload is None else payload, permit)
        finally:
            logger.removeHandler(handler)
        elapsed = time.perf_counter_ns() - started
        after = env.store.counters()
        delta = {k: after[k] - before[k] for k in before}
        session = env.mount.registration.counters()
        self.check(world == env.store.world_hash(), "WORLD_ROW_VERSION_AND_EVENTS_UNCHANGED")
        self.check(delta["forbidden"] == 0, "NO_OLD_DISPATCH_OR_NONALLOWLIST_SQL")
        self.check(public["audit"]["source_operations_delta"] == delta["operations"], "PUBLIC_BILLING_MATCHES_SOURCE_DELTA")
        self.check(delta["select_dispatches"] <= 1 and session["session_claims"] <= 1, "SESSION_READ_LIMIT_ONE")
        rendered = json.dumps(public) + stdout.getvalue() + stderr.getvalue() + logs.getvalue() + repr(env.mount)
        self.verify_public(public, rendered)
        self.check(stdout.getvalue() == stderr.getvalue() == logs.getvalue() == "", "NO_UNCONTROLLED_LOG_OR_STDOUT")
        self.calls.append({"status": public["status"], "category": h.outcome_category(public["status"]),
                           "summary": public, "world_before": world, "world_after": env.store.world_hash(),
                           "store_delta": delta, "session_before": session_before, "session_after": session,
                           "source_stage_count": len(private_stage), "elapsed_ns": elapsed, "secret_scan": "CLEAN"})
        return public, private_stage

    def test_A09_multiple_permits(self):
        env = self.environment()
        permits = [env.mount.issue_permit() for _ in range(3)]
        barrier = threading.Barrier(3)
        before = env.store.world_hash()
        def one(permit):
            worker = threading.get_ident()
            barrier.wait(3)
            return worker, a.dispatch_authority_read(env.mount, env.peer, env.mount.refs(permit), permit)
        with ThreadPoolExecutor(max_workers=3) as executor:
            results = list(executor.map(one, permits))
        statuses = [result[1]["status"] for result in results]
        self.check(statuses.count("STAGING_READY_WITH_GAPS") == 1 and statuses.count("HOST_PERMIT_REPLAY") == 2,
                   "THREE_PERMITS_SHARE_ONE_ATOMIC_SLOT")
        self.check(env.store.select_dispatches == env.store.key_calls == env.store.decode_calls == 1, "ONE_SOURCE_READ_ACROSS_THREADS")
        self.check(env.store.world_hash() == before, "CONCURRENT_WORLD_UNCHANGED")
        for _, public in results:
            self.verify_public(public, json.dumps(public))
        try:
            env.mount.issue_permit()
        except h.HostReadError as error:
            self.check(error.code == "HOST_PERMIT_REPLAY", "NEW_PERMIT_AFTER_READ_REFUSED")
        else:
            self.check(False, "NEW_PERMIT_MUST_NOT_REARM")
        self.batches.append({"kind": "three_permit_race", "thread_ids": sorted({x[0] for x in results}),
                             "statuses": statuses, "world_before": before, "world_after": env.store.world_hash(),
                             "session": env.mount.registration.counters()})
        self.denied_zero(env, "HOST_PERMIT_REPLAY", permit=permits[0])
        # Recreating a trusted issuer must not reset the same process/source
        # authorization namespace, even if a display source alias changes.
        duplicate_issuer = a.ActivationIssuer(env.issuer.epoch_ref, clock=lambda: env.clock.now)
        duplicate_ticket = duplicate_issuer.authorize_offline(authorization_id=env.authorization,
            scope=self.seed["scope_A"], grant_ref="gA", rid="rF", manager_identity="h2_reader",
            principal_ref="manager:a1", actor="user", transport_ref="rpc:authority", scope_ref="scope:a1",
            producer_ref="producer:a1", source_ref="authority:alias", expires_ns=1000000000,
            expected_store=env.store)
        try:
            duplicate = a.mount_authority_host(env.store, duplicate_ticket)
        except h.HostReadError:
            self.check(env.store.select_dispatches == 1, "ISSUER_RECREATION_CANNOT_RESET_SOURCE_SESSION")
        else:
            peer = a.authenticate_authority_peer(duplicate, "Bearer " + self.canaries["rpc_token"],
                                                 env.identities, env.permissions)
            escaped = a.dispatch_authority_read(duplicate, peer, duplicate.refs())
            self.batches.append({"kind": "issuer_recreation_escape", "status": escaped["status"],
                                 "same_authorization_ref": env.authorization,
                                 "total_source_selects": env.store.select_dispatches})
            self.check(False, "ISSUER_RECREATION_CANNOT_RESET_SOURCE_SESSION")

    def test_A18_client(self):
        env = self.environment()
        path = Path(__file__).parents[1] / "scripts/read_dsat_authority_once.py"
        spec = importlib.util.spec_from_file_location("a1_offline_client", path)
        client = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(client)
        transport = Transport(env)
        manifest = {"schema": "dsat-a1-offline-client/1", "mode": "OFFLINE_A1", "payload": env.mount.refs()}
        invocation = client.OfflineInvocation(transport)
        public = json.loads(invocation.run(manifest))
        self.verify_public(public, json.dumps(public))
        self.check(transport.calls == 1 and env.store.select_dispatches == 1, "CLIENT_ONE_SYNTHETIC_CALL")
        try:
            invocation.run(manifest)
        except client.ClientRefusal:
            pass
        else:
            self.check(False, "CLIENT_CANNOT_RETRY")
        for argv in ([], ["--mode", "real", "--manifest", "unused"], ["--mode", "offline", "--manifest", "unused", "--key", self.canaries["private_key"]]):
            output = io.StringIO()
            with contextlib.redirect_stdout(output):
                code = client.main(argv)
            self.check(code == 2 and all(v not in output.getvalue() for v in self.canaries.values()), "STRICT_ARGV_DEFAULT_CLOSED_AND_SANITIZED")
        output = io.StringIO()
        with contextlib.redirect_stdout(output):
            code = client.main(["--mode", "offline", "--manifest", "unused"])
        self.check(code == 2, "NO_LIVE_TRANSPORT_OR_FILE_READ_DEFAULT")
        output = io.StringIO()
        with contextlib.redirect_stdout(output):
            code = client.main(["--mode", "offline", "--manifest", str(CONFIG["fixture_root"] / "CLIENT_MANIFEST.json")], transport=transport)
        retry = json.loads(output.getvalue())
        self.check(code == 0 and retry["status"] == "HOST_PERMIT_REPLAY" and env.store.select_dispatches == 1,
                   "NEW_CLIENT_INVOCATION_CANNOT_REOPEN_SERVER_SESSION")
        self.verify_public(retry, output.getvalue())
