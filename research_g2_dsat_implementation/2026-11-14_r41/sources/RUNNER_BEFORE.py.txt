def sha(path):

        return ENTRY_STATE.hash_owned_file(Path(path))
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def metadata(path, operation):
    require(operation in {"stat", "lstat", "exists", "is_file", "is_dir", "is_symlink"},
        "METADATA_OPERATION_REFUSED")
    read_gate(path, "metadata")
    value = getattr(path, operation)()
    read_gate(path, "metadata_return")
    return value


def descriptor_metadata(stream, path, operation):
    read_gate(path, "metadata")
    require(operation in {"fstat", "tell"}, "METADATA_OPERATION_REFUSED")
    value = os.fstat(stream.fileno()) if operation == "fstat" else stream.tell()
    read_gate(path, "metadata_return")
    return value


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=True,
                      allow_nan=False).encode("utf-8")


def json_size(value, limit, *, text_limit=MAX_TEXT_JSON_BYTES):
    """Validate exact builtin JSON data and count escaped bytes before encoding.

    No repr/str hooks, iterators on user containers, codec, or source reads.
    A rejected value is never replaced by a purported lossless summary.
    """
    nodes = 0
    def walk(item, depth):
        nonlocal nodes
        nodes += 1
        require(nodes <= MAX_JSON_NODES and depth <= MAX_JSON_DEPTH, "EVIDENCE_SHAPE_LIMIT")
        kind = type(item)
        if item is None: return 4
        if kind is bool: return 4 if item else 5
        if kind is int:
            require(item.bit_length() <= MAX_INTEGER_BITS, "EVIDENCE_INTEGER_LIMIT")
            return len(str(item))
        if kind is float:
            require(float("-inf") < item < float("inf"), "EVIDENCE_NUMBER_LIMIT")
            return len(json.dumps(item, allow_nan=False))
        if kind is str:
            # Each non-BMP code point occupies two ASCII Unicode escapes.
            require(len(item) <= text_limit, "EVIDENCE_TEXT_LIMIT")
            size = 2
            for char in item:
                cp = ord(char)
                size += (2 if char in '\"\\\\\b\f\n\r\t' else
                         6 if cp < 32 or 127 <= cp <= 65535 else 12 if cp > 65535 else 1)
                require(size <= text_limit, "EVIDENCE_TEXT_LIMIT")
            return size
        require(kind in (dict, list, tuple), "EVIDENCE_TYPE_LIMIT")
        require(len(item) <= MAX_CONTAINER_ITEMS, "EVIDENCE_CONTAINER_LIMIT")
        size = 2 + max(0, len(item) - 1)
        if kind is dict:
            for key, child in item.items():
                require(type(key) is str, "EVIDENCE_KEY_LIMIT")
                size += walk(key, depth + 1) + 1 + walk(child, depth + 1)
                require(size <= limit, "EVIDENCE_PAYLOAD_LIMIT")
        else:
            for child in item:
                size += walk(child, depth + 1)
                require(size <= limit, "EVIDENCE_PAYLOAD_LIMIT")
        return size
    size = walk(value, 0)
    require(size <= limit, "EVIDENCE_PAYLOAD_LIMIT")
    return size


def bounded_canonical(value, limit=MAX_RAW_JSON_BYTES, *, text_limit=MAX_TEXT_JSON_BYTES):
    expected = json_size(value, limit, text_limit=text_limit)
    data = canonical(value)
    require(len(data) == expected, "EVIDENCE_SERIALIZER_MISMATCH")
    return data


def builtin_type_field(value, name):
    """Use type's descriptors directly; no custom metaclass formatting."""
    result = type.__dict__[name].__get__(type(value), type(type(value)))
    require(type(result) is str, "EVIDENCE_TYPE_LIMIT")
    return result


def module_dictionary(module):
    """Read the builtin module dictionary without a subclass attribute hook."""
    if isinstance(module, type(sys)):
        return type(sys).__dict__["__dict__"].__get__(module, type(module))
    return {}


def load(path):
    path = checked(path)
    require(path.stat().st_size <= 4 * 1024 * 1024, "CONTROL_SIZE_LIMIT")
    return json.loads(path.read_text(encoding="utf-8-sig"))


def checked(path, *, output=False, historical_read=False):
    raw = str(path)
    require(Path(raw).is_absolute() and not raw.startswith(("\\\\", "//"))
        and ":" not in raw[2:] and not any(p in {".", ".."} or p.endswith((".", " "))
        for p in raw.replace("/", "\\").split("\\")[1:]), "PATH_DENIED")
    p = Path(raw).absolute()
    if output:
        require(p == OUTPUT or p == RUN or (p.is_relative_to(OUTPUT)
            and p.relative_to(OUTPUT).as_posix() in WRITE_NAMES), "OUTPUT_PATH_DENIED")
    for part in (p, *p.parents):
        require(not metadata(part, "is_symlink") and not (metadata(part, "exists")
            and getattr(metadata(part, "lstat"), "st_file_attributes", 0) & 0x400), "REPARSE_DENIED")
    if metadata(p, "is_file"):
        require(metadata(p, "stat").st_nlink == 1 or (historical_read and not output
            and p in HISTORICAL_READ_ONLY_LINKS), "HARDLINK_DENIED")
    return p

    def begin_terminal(self):
        self.terminal = True
        self.phase = "TERMINAL"
        if time.perf_counter() >= self.terminal_deadline:
            self.finalization_incomplete = True

    def terminal_read_gate(self, path, operation):
        require(self.terminal, "TERMINAL_READ_PHASE_REQUIRED")
        if time.perf_counter() >= min(self.terminal_read_deadline, self.deadline):
            self.read_audit["denied_terminal_checks"] += 1
            self.read_audit["last_denial"] = "TERMINAL_READ_DEADLINE"
            if operation.endswith("return"): self.read_audit["late_return_observations"] += 1
            self.finalization_incomplete = True
            self.incomplete("TERMINAL_READ_DEADLINE")
            raise Refusal("TERMINAL_READ_DEADLINE")
        path = Path(path)
        owned_file = path.is_absolute() and path.is_relative_to(OUTPUT) and (
            path.relative_to(OUTPUT).as_posix() in WRITE_NAMES)
        directory = path in (OUTPUT, RUN)
        ancestor_metadata = operation in {"metadata", "metadata_return"} and OUTPUT.is_relative_to(path)
        if operation.startswith("directory_"):
            allowed = directory
        elif operation.startswith("metadata"):
            allowed = owned_file or directory or ancestor_metadata
        else:
            allowed = owned_file
        if not allowed:
            self.read_audit["denied_terminal_checks"] += 1
            self.read_audit["last_denial"] = "TERMINAL_READ_SCOPE_REFUSED"
            self.finalization_incomplete = True
            self.incomplete("TERMINAL_READ_SCOPE_REFUSED")
            raise Refusal("TERMINAL_READ_SCOPE_REFUSED")
        self.read_audit["admitted_terminal_checks"] += 1

    def hash_owned_file(self, path):
        read_gate(path, "content_open")
        path = checked(path, output=True)
        require(len(self.read_handles) < MAX_TERMINAL_READ_HANDLES, "TERMINAL_READ_HANDLE_LIMIT")
        item = {"path": path, "stream": None, "close_attempted": False}
        self.read_handles.append(item)
        digest = hashlib.sha256()
        total = 0
        failed = False
        try:
            read_gate(path, "content_open")
            item["stream"] = path.open("rb", buffering=0)
            # Opening can block:admit each subsequent read separately.
            read_gate(path, "content_return")
            cap = EVIDENCE_CAPS[path.relative_to(OUTPUT).as_posix()]
            while True:
                read_gate(path, "content_read")
                data = item["stream"].read(65536)
                read_gate(path, "content_return")
                if not data: break
                total += len(data)
                require(total <= cap, "OWNED_READ_EXTENT_LIMIT")
                digest.update(data)
            read_gate(path, "hash_return")
            return digest.hexdigest()
        except BaseException:
            failed = True
            self.finalization_incomplete = True
            self.incomplete("TERMINAL_READ_INCOMPLETE")
            raise
        finally:
            try: self.close_read_handles(item)
            except BaseException as error:
                self.failure("CLEANUP_READ_HANDLE", error)
                if not failed: raise

    def close_read_handles(self, only=None):
        for item in self.read_handles if only is None else [only]:
            stream = item["stream"]
            if stream is None or stream.closed or item["close_attempted"]: continue
            if time.perf_counter() >= self.deadline:
                self.finalization_incomplete = True
                self.incomplete("READ_HANDLE_CLOSE_UNCONFIRMED")
                continue
            item["close_attempted"] = True
            try:
                stream.close()
                require(stream.closed, "READ_HANDLE_CLOSE_UNCONFIRMED")
                self.read_audit["confirmed_closed_read_handles"] += 1
                if time.perf_counter() >= self.deadline:
                    self.finalization_incomplete = True
                    self.incomplete("LATE_READ_CLOSE_COMPLETION")
            except BaseException:
                self.finalization_incomplete = True
                self.persistence_ok = False
                self.incomplete("READ_HANDLE_CLOSE_UNCONFIRMED")
                raise

        directories = [OUTPUT]
        for directory in directories:
            read_gate(directory, "directory_open")
            names = os.listdir(directory)
            read_gate(directory, "directory_return")
            require(len(names) <= 14, "RESOURCE_FILE_COUNT")
            for name in sorted(names):
                read_gate(directory, "directory_next")
                path = directory / name
                if path == RUN:
                    require(directory == OUTPUT and metadata(checked(path, output=True), "is_dir"),
                        "OUTPUT_DIRECTORY_IDENTITY")
                    directories.append(RUN)
                    continue
                require(path.relative_to(OUTPUT).as_posix() in WRITE_NAMES, "OUTPUT_PATH_DENIED")
                path = checked(path, output=True)
                require(metadata(path, "is_file"), "OUTPUT_FILE_IDENTITY")
                files.append(path)
                require(len(files) <= 13, "RESOURCE_FILE_COUNT")
        return files

    def evidence_text(self, value, maximum=MAX_METADATA_JSON_BYTES):
        require(type(value) is str, "EVIDENCE_TYPE_LIMIT")
        # Redaction is complete before any text enters the public object.
        require(len(value) <= maximum, "EVIDENCE_TEXT_LIMIT")
        clean = self.redact(value)
        try:
            json_size(clean, maximum, text_limit=maximum)
        except Refusal:
            self.incomplete("EVIDENCE_CONTRACT_UNSATISFIED")
            raise
        return clean

    def exception_record(self, error):
        """Bounded descriptor reads; arbitrary __str__/__repr__ are never called.

        Full original formatting remains an outstanding evidence obligation.
        Every diagnostic descriptor therefore fails complete acceptance.
        """
        self.incomplete("EVIDENCE_CONTRACT_UNSATISFIED")
        record = {"exception_type": "UNKNOWN", "exception_module": "UNKNOWN",
            "errno": None, "winerror": None, "text": "UNRENDERED",
            "FULL_EXCEPTION_TEXT_PRESERVED": False, "original_text_bytes": "UNKNOWN",
            "original_text_sha256": "UNKNOWN", "evidence_incomplete": True,
            "traceback": [], "chain": [], "original_frame_count": "UNKNOWN",
            "source_excerpt_bytes": 0}
        def text(value, cap=1024):
            try: return self.evidence_text(value, cap)
            except Refusal: return "UNAVAILABLE_OVER_BOUND"
        record["exception_type"] = text(builtin_type_field(error, "__name__"), 256)
        record["exception_module"] = text(builtin_type_field(error, "__module__"), 256)
        os_fields_safe = True
        if isinstance(error, OSError):
            for name in ("errno", "winerror"):
                desc = OSError.__dict__.get(name)
                value = desc.__get__(error, type(error)) if desc is not None else None
                record[name] = value if type(value) is int and value.bit_length() <= 64 else None
            for name in ("filename", "filename2", "strerror"):
                desc = OSError.__dict__.get(name)
                value = desc.__get__(error, type(error)) if desc is not None else None
                record[name] = text(value) if type(value) is str else None
                if value is not None:
                    try:
                        require(type(value) is str, "EVIDENCE_TYPE_LIMIT")
                        json_size(value, 1024, text_limit=1024)
                    except Refusal: os_fields_safe = False
        args = BaseException.__dict__["args"].__get__(error, type(error))
        try:
            json_size(args, MAX_EXCEPTION_ARGS_BYTES, text_limit=1024)
            # Exact builtin identities only; Refusal inherits builtin formatting.
            cls = type(error)
            trusted = cls is Refusal or any(cls is vars(builtins).get(name) for name in
                ("Exception", "RuntimeError", "ValueError", "TypeError", "AssertionError",
                 "OSError", "PermissionError", "FileNotFoundError", "FileExistsError",
                 "TimeoutError", "InterruptedError", "KeyboardInterrupt", "MemoryError"))
            if trusted and os_fields_safe:
                rendered = cls.__str__(error)
                record["text"] = text(rendered, 4096)
            record["args"] = args
        except (Refusal, ValueError, TypeError):
            record["args"] = "UNAVAILABLE_OVER_BOUND_OR_NONPRIMITIVE"
        trace = BaseException.__dict__["__traceback__"].__get__(error, type(error))
        while trace is not None and len(record["traceback"]) < MAX_TRACEBACK_FRAMES:
            code = trace.tb_frame.f_code
            record["traceback"].append({"file": text(code.co_filename),
                "function": text(code.co_name, 256), "module": text(
                    trace.tb_frame.f_globals.get("__name__", "UNKNOWN"), 256),
                "line": trace.tb_lineno if trace.tb_lineno.bit_length() <= 64 else "UNKNOWN"})
            trace = trace.tb_next
        record["traceback_omitted"] = trace is not None
        if trace is None: record["original_frame_count"] = len(record["traceback"])
        current, seen = error, {id(error)}
        for _ in range(MAX_EXCEPTION_CHAIN):
            cause = BaseException.__dict__["__cause__"].__get__(current, type(current))
            context = BaseException.__dict__["__context__"].__get__(current, type(current))
            child = cause if cause is not None else context
            if child is None: break
            if id(child) in seen:
                record["chain"].append({"cycle": True}); break
            seen.add(id(child))
            record["chain"].append({"relation": "cause" if cause is not None else "context",
                "exception_type": text(builtin_type_field(child, "__name__"), 256),
                "text": "UNRENDERED", "FULL_EXCEPTION_TEXT_PRESERVED": False})
            current = child
        else: record["chain_omitted_or_unknown"] = True
        # Validate the descriptor, including its byte bound, before serialization.
        try: json_size(record, MAX_EXCEPTION_JSON_BYTES)
        except Refusal:
            return {"exception_type": record["exception_type"],
                "exception_module": record["exception_module"], "errno": record["errno"],
                "winerror": record["winerror"], "FULL_EXCEPTION_TEXT_PRESERVED": False,
                "evidence_incomplete": True, "descriptor_overflow": True,
                "original_text_bytes": "UNKNOWN", "original_text_sha256": "UNKNOWN"}
        return record

    def encoded_payload(self, value):
        """Lossless representation; validate the public sink before compression."""
        # Prepared while WORK is active; terminal never performs a lazy import.
        require(self.codec is not None, "CODEC_NOT_PREPARED")
        binascii, zlib = self.codec
        raw = bounded_canonical(value)
        require(all(v.encode() not in raw for v in self.secrets), "PUBLIC_SINK_REFUSED")
        compressed = zlib.compress(raw, level=9)
        require(len(compressed) <= MAX_COMPRESSED_BYTES, "EVIDENCE_CODEC_LIMIT")
        return {"encoding": "zlib-base64-canonical-json/1", "raw_bytes": len(raw),
            "raw_sha256": hashlib.sha256(raw).hexdigest(),
            "data": binascii.b2a_base64(compressed, newline=False).decode("ascii")}

    def case_error_header(self, error):
        if error is None: return None
        header = {"exception_type": "UNKNOWN", "exception_module": "UNKNOWN",
            "errno": "NOT_APPLICABLE", "winerror": "NOT_APPLICABLE",
            "known_refusal_code": None, "header_complete": True}
        for key, attribute in (("exception_type", "__name__"), ("exception_module", "__module__")):
            try: header[key] = self.evidence_text(builtin_type_field(error, attribute), 256)
            except BaseException:
                header["header_complete"] = False
                self.incomplete("RESULT_HEADER_INCOMPLETE")
        if isinstance(error, OSError):
            for key in ("errno", "winerror"):
                try:
                    descriptor = OSError.__dict__.get(key)
                    value = descriptor.__get__(error, type(error)) if descriptor is not None else None
                    require(value is None or (type(value) is int and value.bit_length() <= 64),
                        "RESULT_HEADER_INTEGER")
                    header[key] = value
                except BaseException:
                    header[key] = "UNKNOWN"
                    header["header_complete"] = False
                    self.incomplete("RESULT_HEADER_INCOMPLETE")
        if type(error) is Refusal:
            args = BaseException.__dict__["args"].__get__(error, type(error))
            try:
                require(len(args) == 1 and type(args[0]) is str, "RESULT_HEADER_CODE")
                header["known_refusal_code"] = self.evidence_text(args[0], 256)
            except BaseException:
                header["known_refusal_code"] = "UNKNOWN"
                header["header_complete"] = False
                self.incomplete("RESULT_HEADER_INCOMPLETE")
        return header

    def commit_case_record(self, record, stage):
        # A record receives at most one attempt;an acknowledgement is immutable.
        if record["persistence_state"] != "RESULT_OBSERVED_IN_MEMORY":
            return record["persistence_state"] == "RESULT_CHECKPOINT_COMMITTED"
        record.update(persistence_state="RESULT_PERSISTENCE_UNKNOWN", RESULT_DURABLE="UNKNOWN")
        key = RUN_ID + "/EARLY_FAILURE.json"
        before = self.byte_ledger.get(key, {}).get("WRITE_ATTEMPTED_BYTES", 0)
        try:
            reference = self.checkpoint(stage, record["observed"])
        except BaseException as error:
            attempted = self.byte_ledger.get(key, {}).get("WRITE_ATTEMPTED_BYTES", 0) > before
            record.update(persistence_state="RESULT_PERSISTENCE_UNKNOWN" if attempted else "RESULT_PERSISTENCE_FAILED",
                RESULT_DURABLE="UNKNOWN" if attempted else False)
            self.persistence_ok = False
            self.incomplete("CASE_CHECKPOINT_UNCONFIRMED")
            if key in self.byte_ledger: self.byte_ledger[key]["poisoned"] = True
            self.failure("PERSIST_" + stage, error)
            return False
        record.update(checkpoint_ref=reference, persistence_state="RESULT_CHECKPOINT_COMMITTED", RESULT_DURABLE=True)
        return True

    def begin_case(self, tid):
        self.check_work()
        spec = self.spec_by_id[tid]
        require(spec["test_source"] == self.phase and self.case_lifecycle[tid] == "NOT_RUN",
            "CASE_START_IDENTITY")
        # Preallocated lifecycle survives a later allocation/diagnostic failure.
        self.case_lifecycle[tid] = "RUNNING"
        row = {"id": tid, "group": spec["test_source"], "status": "RUNNING", "executed": True,
            "start_boundary": "RESULT_STARTTEST_ENTERED",
            "trusted_execution_phase": self.phase, "source_hash": spec["source_hash"],
            "runner_sha256": self.command["runner_sha256"], "original_expected": spec["original_expected"],
            "events": [], "detail": None, "diagnostic_status": "NOT_APPLICABLE",
            "details_status": "NOT_CAPTURED",
            "detail_indices": [], "case_call_completed": False, "case_interrupted": False}
        self.rows[tid] = row
        record = {"observed": {"schema": "r38-case-start/1", "id": tid,
            "group": row["group"], "status": "RUNNING", "source_hash": row["source_hash"],
            "runner_sha256": row["runner_sha256"], "trusted_execution_phase": self.phase},
            "persistence_state": "RESULT_OBSERVED_IN_MEMORY", "RESULT_DURABLE": False,
            "checkpoint_ref": None}
        row["start_record"] = record
        require(self.commit_case_record(record, "TEST_CASE_STARTED"), "CASE_START_CHECKPOINT_FAILED")

    def observe_result(self, tid, status, elapsed_ns, error=None):
        require(tid in self.rows and status in {"PASS", "FAIL", "ERROR", "SKIP"}, "CASE_RESULT_IDENTITY")
        index = self.case_callback_counts[tid]
        if index >= MAX_CASE_RESULT_EVENTS:
            self.case_overflow[tid] = True
            self.incomplete("CASE_OUTCOME_COUNT_EXCEEDED")
            raise Refusal("CASE_OUTCOME_COUNT_EXCEEDED")
        # Preserve the original callback status BEFORE header/JSON/log work.
        self.case_statuses[tid][index] = status
        self.case_callback_counts[tid] = index + 1
        self.case_lifecycle[tid] = "OUTCOME_OBSERVED"
        row = self.rows[tid]
        row["status"] = status if index == 0 else "UNKNOWN"
        if status != "PASS": row["diagnostic_status"] = "ORIGINAL_DIAGNOSTIC_UNCONFIRMED"
        if index:
            self.incomplete("MULTIPLE_ORIGINAL_CASE_OUTCOMES")
        observed = {"schema": "r38-result-observation/1", "id": tid,
            "event_id": tid + ":" + str(index + 1), "status": status,
            "group": row["group"], "source_hash": row["source_hash"],
            "runner_sha256": row["runner_sha256"], "original_expected": row["original_expected"],
            "trusted_execution_phase": row["trusted_execution_phase"], "elapsed_ns": elapsed_ns,
            "exception_category": "ORIGINAL_ASSERTION_FAILURE" if status == "FAIL" else
                "ORIGINAL_TEST_ERROR" if status == "ERROR" else "NOT_APPLICABLE",
            "error_header": self.case_error_header(error)}
        record = {"observed": observed, "persistence_state": "RESULT_OBSERVED_IN_MEMORY",
            "RESULT_DURABLE": False, "checkpoint_ref": None}
        row["events"].append(record)
        require(self.commit_case_record(record, "TEST_ASSERTION"), "RESULT_CHECKPOINT_UNCONFIRMED")
        return record

    def capture_case_details(self, tid, module, test, interrupted):
        row = self.rows[tid]
        values = module_dictionary(module).get("DETAILS")
        require(type(values) is list and len(values) <= 68, "TEST_DETAIL_COUNT_LIMIT")
        group = row["group"]
        cursor = self.detail_cursors.get(group, 0)
        require(cursor <= len(values), "DETAIL_SOURCE_REWOUND")
        new_details = []
        while cursor < len(values):
            value = values[cursor]
            # Exact JSON copy of cached public primitive data;no test method calls.
            frozen = json.loads(bounded_canonical(value, 65536))
            require(type(frozen) is dict and frozen.get("id") in self.rows, "DETAIL_CASE_IDENTITY")
            detail_tid = frozen["id"]
            require(self.rows[detail_tid]["group"] == group and len(self.details) < 68,
                "DETAIL_GROUP_IDENTITY")
            detail_index = len(self.details)
            self.details.append(frozen)
            self.rows[detail_tid]["detail_indices"].append(detail_index)
            self.rows[detail_tid]["details_status"] = "CAPTURED_IN_MEMORY"
            new_details.append({"index": detail_index, "id": detail_tid,
                "raw_sha256": hashlib.sha256(canonical(frozen)).hexdigest(), "value": frozen})
            cursor += 1
            # Advance immediately after memory capture;never duplicate on failure.
            self.detail_cursors[group] = cursor
        partial_checks = None
        if not row["detail_indices"]:
            row["details_status"] = "MISSING_OR_INTERRUPTED"
            self.incomplete("CASE_DETAILS_INCOMPLETE")
            test_data = object.__getattribute__(test, "__dict__")
            checks = test_data.get("checks")
            if type(checks) is list:
                partial_checks = json.loads(bounded_canonical(checks, 65536))
                row["partial_checks"] = partial_checks
        return new_details, partial_checks

    def end_case(self, tid, module, test, interrupted):
        if self.case_lifecycle[tid] == "NOT_RUN": return
        self.case_lifecycle[tid] = "INTERRUPTED" if interrupted else "COMPLETED"
        if tid not in self.rows:
            self.incomplete("CASE_ROW_ALLOCATION_INCOMPLETE")
            return
        row = self.rows[tid]
        row["case_call_completed"] = not interrupted
        row["case_interrupted"] = interrupted
        if not self.case_callback_counts[tid]:
            row["status"] = "RUNNING_INTERRUPTED" if interrupted else "UNKNOWN"
            self.incomplete("CASE_TERMINAL_RESULT_UNKNOWN")
        new_details, partial_checks = self.capture_case_details(tid, module, test, interrupted)
        record = {"observed": {"schema": "r38-case-evidence/1", "id": tid,
            "group": row["group"], "source_hash": row["source_hash"],
            "case_call_completed": row["case_call_completed"], "case_interrupted": interrupted,
            "observed_callback_statuses": [x for x in self.case_statuses[tid] if x is not None],
            "callback_overflow": self.case_overflow[tid],
            "outcome_records": row["events"], "details_status": row["details_status"],
            "new_details": new_details, "partial_checks": partial_checks,
            "diagnostic_display": row["detail"], "diagnostic_status": row["diagnostic_status"]},
            "persistence_state": "RESULT_OBSERVED_IN_MEMORY", "RESULT_DURABLE": False,
            "checkpoint_ref": None}
        # The final case row holds the acknowledgement,not another copy of
        # original DETAILS. Full values remain in self.details and the journal.
        row["end_record"] = {"id": tid, "persistence_state": "RESULT_OBSERVED_IN_MEMORY",
            "RESULT_DURABLE": False, "checkpoint_ref": None}
        try:
            committed = self.commit_case_record(record, "TEST_CASE_EVIDENCE")
        finally:
            row["end_record"].update({key: record[key] for key in
                ("persistence_state", "RESULT_DURABLE", "checkpoint_ref")})
        if committed:
            for detail in new_details:
                self.detail_refs[str(detail["index"])] = record["checkpoint_ref"]
                self.rows[detail["id"]]["details_status"] = "DETAILS_CHECKPOINT_COMMITTED"
        require(committed, "CASE_EVIDENCE_CHECKPOINT_UNCONFIRMED")

    def case_partition(self):
        rows = self.assertion_rows()
        known = {x["id"] for x in rows if len(x["observed_statuses"]) == 1
            and not self.case_overflow[x["id"]]}
        return {"denominator": 68,
            "completion_basis": "original callback outcome;case-call return and persistence are separate",
            "case_counting_basis": "one frozen ID per case;multiple original outcomes remain UNKNOWN in case counts",
            "observed_original_callback_counts": {status: sum(
                value == status for slots in self.case_statuses.values() for value in slots)
                for status in ("PASS", "FAIL", "ERROR", "SKIP")},
            "confirmed_result_set": [x["id"] for x in rows if x["id"] in known],
            "begun_result_unconfirmed_set": [x["id"] for x in rows if x["executed"] and x["id"] not in known],
            "not_started": [x["id"] for x in rows if not x["executed"]],
            "started_without_outcome": [x["id"] for x in rows if x["executed"] and not x["observed_statuses"]],
            "outcomes_observed": [x["id"] for x in rows if x["observed_statuses"]],
            "outcome_unknown_or_multiple": [x["id"] for x in rows if x["status"] == "UNKNOWN"],
            "case_calls_completed": [x["id"] for x in rows if x.get("case_call_completed") is True],
            "case_calls_interrupted": [x["id"] for x in rows if x.get("case_interrupted") is True],
            "durable_outcomes": [x["id"] for x in rows if x["RESULT_DURABLE"] is True],
            "memory_only_or_uncertain_outcomes": [x["id"] for x in rows
                if x["observed_statuses"] and x["RESULT_DURABLE"] is not True]}

    def capture_details(self, values):
        require(type(values) is list and len(self.details) + len(values) <= 68,
            "TEST_DETAIL_COUNT_LIMIT")
        for value in values: json_size(value, 65536)
        self.details.extend(values)

    def checkpoint(self, stage, value):
        """Append-only, precharged FileIO; only fsynced lines advance the head."""
        path = checked(RUN / "EARLY_FAILURE.json", output=True)
        with self.io_lock:
            require(not self.journal_closed, "JOURNAL_ALREADY_CLOSED")
            require(self.checkpoint_attempts < MAX_CHECKPOINT_ATTEMPTS, "CHECKPOINT_ATTEMPT_LIMIT")
            self.checkpoint_attempts += 1
            event = {"schema": "r35-checkpoint-jsonl/1", "seq": self.journal_sequence + 1,
                "previous_sha256": self.journal_head, "stage": self.evidence_text(stage, 256),
                "phase": self.phase, "test_id": self.current,
                "utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
                "value": self.encoded_payload(value)}
            raw = bounded_canonical(event, EVIDENCE_CAPS[RUN_ID + "/EARLY_FAILURE.json"],
                text_limit=2 * MAX_RAW_JSON_BYTES)
            digest = hashlib.sha256(raw).hexdigest()
            data = bounded_canonical({**event, "sha256": digest},
                EVIDENCE_CAPS[RUN_ID + "/EARLY_FAILURE.json"] - 1,
                text_limit=2 * MAX_RAW_JSON_BYTES) + b"\n"
            require(all(v.encode() not in data for v in self.secrets), "PUBLIC_SINK_REFUSED")
            self.reserve_bytes(len(data), path)
            if self.journal is None:
                self.journal = self.open_owned(path)
                stat = descriptor_metadata(self.journal, path, "fstat")
                self.journal_identity = (stat.st_dev, stat.st_ino)
            stat = metadata(path, "stat")
            require((stat.st_dev, stat.st_ino) == self.journal_identity
                and stat.st_size == self.journal_bytes
                and descriptor_metadata(self.journal, path, "tell") == self.journal_bytes, "JOURNAL_IDENTITY_CHANGED")
            self.io_write(self.journal, path, data)
            self.journal_sequence += 1
            self.journal_head = digest
            self.journal_bytes += len(data)
            return {"seq": self.journal_sequence, "sha256": digest,
                "stage": stage, "test_id": self.current, "path": str(path)}

    def failure(self, stage, error):
        code = "UNKNOWN_EXCEPTION"
        if type(error) is Refusal:
            args = BaseException.__dict__["args"].__get__(error, type(error))
            if len(args) == 1 and type(args[0]) is str:
                try: code = self.evidence_text(args[0], 256)
                except Refusal: code = "EVIDENCE_CONTRACT_UNSATISFIED"
        try:
            descriptor = self.exception_record(error)
        except BaseException:
            self.incomplete("DIAGNOSTIC_RECORDER_FAILED")
            descriptor = {"exception_type": "UNKNOWN", "FULL_EXCEPTION_TEXT_PRESERVED": False,
                "evidence_incomplete": True, "recorder_failed": True,
                "original_text_bytes": "UNKNOWN", "original_text_sha256": "UNKNOWN"}
        if stage.startswith(("PERSIST", "CHECKPOINT")): category = "OUTPUT_PERSISTENCE_ERROR"
        elif code == "DEPENDENCY_CHANGED": category = "FILE_DIGEST_ERROR"
        elif isinstance(error, PermissionError) or code == "READ_SCOPE_REFUSED":
            category = "READ_PERMISSION_ERROR"
        elif code in {"DEPENDENCY_ORIGIN_MISMATCH", "MODULE_ORIGIN_MISMATCH"}:
            category = "DEPENDENCY_PROVENANCE_ERROR"
        elif code == "ORIGIN_UNRESOLVED": category = "ORIGIN_INFORMATION_INSUFFICIENT"
        else: category = "AUDIT_OR_EXECUTION_ERROR"
        row = {"stage": self.evidence_text(stage, 256), "code": code,
            "category": category, "exception": descriptor}
        if len(self.errors) < MAX_FAILURE_RECORDS: self.errors.append(row)
        else: self.incomplete("FAILURE_RECORD_LIMIT")
        return row

    def safe_checkpoint(self, stage, value):
        try:
            self.checkpoint(stage, value)
            return True
        except BaseException as error:
            self.persistence_ok = False
            key = RUN_ID + "/EARLY_FAILURE.json"
            if key in self.byte_ledger: self.byte_ledger[key]["poisoned"] = True
            self.failure("CHECKPOINT_" + stage, error)
            return False

    def assertion_rows(self):
        rows = []
        for spec in self.manifest["tests"]:
            tid = spec["test_id"]
            started = self.case_lifecycle[tid] != "NOT_RUN"
            statuses = [x for x in self.case_statuses[tid] if x is not None]
            status = ("UNKNOWN" if len(statuses) > 1 or self.case_overflow[tid] else statuses[0]) if statuses else (
                "RUNNING_INTERRUPTED" if self.case_lifecycle[tid] == "INTERRUPTED" else
                "UNKNOWN" if self.case_lifecycle[tid] == "COMPLETED" else "RUNNING" if started else "NOT_RUN")
            row = {**{k: spec[k] for k in ("test_source", "test_callable", "original_expected", "source_hash")},
                **self.rows.get(tid, {"id": tid, "group": spec["test_source"]}), "status": status, "executed": started,
                "case_call_completed": self.case_lifecycle[tid] == "COMPLETED",
                "case_interrupted": self.case_lifecycle[tid] == "INTERRUPTED",
                "observed_statuses": statuses}
            records = row.get("events", [])
            row["RESULT_DURABLE"] = bool(records) and len(records) == len(statuses) and all(
                x["RESULT_DURABLE"] is True for x in records)
            if records and any(x["RESULT_DURABLE"] == "UNKNOWN" for x in records):
                row["RESULT_DURABLE"] = "UNKNOWN"
            if statuses and not records: row["RESULT_DURABLE"] = "UNKNOWN"
            rows.append(row)
        return rows

    def assertion_counts(self):
        rows = self.assertion_rows()
        return {key: sum(x["status"] == key for x in rows) for key in
            ("PASS", "FAIL", "ERROR", "SKIP", "BLOCKED", "NOT_RUN", "RUNNING", "RUNNING_INTERRUPTED", "UNKNOWN")}

    def finish_assertions(self):
        if not self.tests_finished:
            self.checkpoint("TESTS_FINISHED", {"counts": self.assertion_counts(), "case_partition": self.case_partition(),
                "cases": self.assertion_rows(), "executed_order": list(self.rows),
                "details": self.details, "projections": self.projections, "finalized": False})
            self.tests_finished = True

    def publish(self, name, value):
        """Each artifact has one exclusive write attempt; other emissions continue."""
        path = OUTPUT / name if name == "TEST_RUNS.json" else RUN / name
        try:
            self.write(path, value)
            self.artifacts[name] = {"path": str(path), "sha256": sha(path)}
            return True
        except BaseException as error:
            self.persistence_ok = False
            row = self.failure("PERSIST_" + name, error)
            self.safe_checkpoint("FAILED", row)
            return False

    def close_journal(self):
        if self.journal_closed: return
        self.journal_closed = True
        self.close_owned(self.journal, RUN / "EARLY_FAILURE.json")

    def refuse(self, code):
        self.stop_requested = True
        if len(self.meter["denials"]) < 64: self.meter["denials"].append(self.evidence_text(code, 256))
        else: self.incomplete("DENIAL_RECORD_LIMIT")
        raise Refusal(code)

    def redact(self, text):
        for value in sorted(self.secrets, key=len, reverse=True):
            text = text.replace(value, "[REDACTED_SYNTHETIC_CANARY]")
        return text

    def sizes(self):
        return sum(metadata(p, "stat").st_size for p in self.output_files())

    def reserve_bytes(self, size, path):
        """Debit BEFORE I/O. Failed, short and uncertain attempts are never refunded."""
        cutoff = self.terminal_deadline if self.terminal else self.work_deadline
        if time.perf_counter() >= cutoff:
            self.finalization_incomplete = True
            raise Refusal("FINALIZATION_INCOMPLETE" if self.terminal else "WORK_DEADLINE_OR_STOP")
        p = checked(path, output=True)
        key = p.relative_to(OUTPUT).as_posix()
        require(key in EVIDENCE_CAPS and key != APPROVAL.name, "OUTPUT_PATH_DENIED")
        if self.terminal:
            require(self.terminal_attempts < TERMINAL_WRITE_ATTEMPTS, "TERMINAL_ATTEMPT_LIMIT")
            self.terminal_attempts += 1
        row = self.byte_ledger.setdefault(key, {"RESERVED_BYTES": EVIDENCE_CAPS[key],
            "SERIALIZED_BYTES": 0, "BUFFERED_BYTES": 0, "WRITE_ATTEMPTED_BYTES": 0,
            "WRITE_RETURNED_BYTES": 0, "CONFIRMED_PERSISTED_BYTES": 0,
            "PERSISTENCE_UNCERTAIN_BYTES": 0, "charged_bytes": 0,
            "write_attempts": 0, "close_attempts": 0, "closed_confirmed": False,
            "state": "RESERVED", "poisoned": False})
        require(not row["poisoned"], "EVIDENCE_STREAM_POISONED")
        cap = EVIDENCE_CAPS[key]
        if p.name == "EARLY_FAILURE.json" and not self.terminal: cap -= TERMINAL_JOURNAL_BYTES
        require(type(size) is int and size >= 0 and row["charged_bytes"] + size <= cap,
            "EVIDENCE_BUDGET_INSUFFICIENT")
        # External mutation is a failed ownership contract, never free space.
        visible = metadata(p, "stat").st_size if metadata(p, "exists") else 0
        require(visible <= row["charged_bytes"], "UNACCOUNTED_EVIDENCE_BYTES")
        row["SERIALIZED_BYTES"] += size
        row["charged_bytes"] += size
        row["state"] = "SERIALIZED"
        require(sum(x["charged_bytes"] for x in self.byte_ledger.values())
            + EVIDENCE_CAPS[APPROVAL.name] <= 2097152, "RESOURCE_OUTPUT_BYTES")
        return row

    def io_write(self, stream, path, data):
        row = self.byte_ledger[path.relative_to(OUTPUT).as_posix()]
        row["state"] = "WRITE_STARTED"
        try:
            cutoff = self.terminal_deadline if self.terminal else self.work_deadline
            require(time.perf_counter() < cutoff, "FINALIZATION_INCOMPLETE")
            row["WRITE_ATTEMPTED_BYTES"] += len(data)
            row["write_attempts"] += 1
            count = stream.write(data)
            if type(count) is int and 0 <= count <= len(data): row["WRITE_RETURNED_BYTES"] += count
            require(count == len(data), "EVIDENCE_SHORT_WRITE")
            # FileIO has no Python pending write buffer. flush is still checked.
            require(time.perf_counter() < cutoff, "FINALIZATION_INCOMPLETE")
            stream.flush()
            row["state"] = "FLUSHED"
            require(time.perf_counter() < cutoff, "FINALIZATION_INCOMPLETE")
            os.fsync(stream.fileno())
            row["CONFIRMED_PERSISTED_BYTES"] = row["WRITE_RETURNED_BYTES"]
            row["state"] = "PUBLISHED"
        except BaseException:
            row["poisoned"] = True
            row["state"] = "PERSISTENCE_UNCERTAIN"
            row["PERSISTENCE_UNCERTAIN_BYTES"] = row["charged_bytes"] - row["CONFIRMED_PERSISTED_BYTES"]
            self.persistence_ok = False
            self.stop_requested = True
            self.finalization_incomplete = True
            raise

    def open_owned(self, path):
        with self.io_lock:
            require(self.opening_output is None, "OUTPUT_OPEN_REENTRY")
            require(path not in self.owned_handles, "OUTPUT_OPEN_RETRY_REFUSED")
            cutoff = self.terminal_deadline if self.terminal else self.work_deadline
            require(time.perf_counter() < cutoff, "FINALIZATION_INCOMPLETE")
            self.opening_output = path
            try:
                stream = path.open("xb", buffering=0)
                self.owned_handles[path] = stream
                return stream
            finally: self.opening_output = None

    def close_owned(self, stream, path):
        if stream is None or stream.closed: return
        if path in self.close_attempted: return
        if time.perf_counter() >= self.deadline:
            self.persistence_ok = False
            self.finalization_incomplete = True
            return
        self.close_attempted.add(path)
        row = self.byte_ledger[path.relative_to(OUTPUT).as_posix()]
        row["close_attempts"] += 1
        try:
            # Unbuffered close cannot emit additional, uncharged payload bytes.
            stream.close()
            row["closed_confirmed"] = stream.closed
            row["state"] = "FINALIZED"
        except BaseException:
            self.persistence_ok = False
            self.finalization_incomplete = True
            self.stop_requested = True
            raise

    def write(self, path, value, *, raw=False):
        p = checked(path, output=True)
        require(p != APPROVAL, "APPROVAL_READ_ONLY")
        key = p.relative_to(OUTPUT).as_posix()
        with self.io_lock:
            require(key not in self.write_attempted_paths, "EVIDENCE_RETRY_REFUSED")
            self.write_attempted_paths.add(key)
            if raw:
                require(p.name in {x[0] for x in ALLOWED_D24_INPUTS}, "RAW_OUTPUT_REFUSED")
                data = value
            elif p.name in ENCODED_OUTPUTS:
                envelope = {"schema": "r35-lossless-artifact/1", "content": self.encoded_payload(value)}
                if p.name == "TEST_RUNS.json":
                    envelope.update(status=value["status"], FULL_CONTRACT_ACCEPTED=value["FULL_CONTRACT_ACCEPTED"])
                data = bounded_canonical(envelope, EVIDENCE_CAPS[key] - 1,
                    text_limit=2 * MAX_RAW_JSON_BYTES) + b"\n"
            else:
                data = bounded_canonical(value, EVIDENCE_CAPS[key] - 1) + b"\n"
            require(type(data) is bytes and all(v.encode() not in data for v in self.secrets), "PUBLIC_SINK_REFUSED")
            self.reserve_bytes(len(data), p)
            stream = None
            try:
                stream = self.open_owned(p)
                self.io_write(stream, p, data)
            except BaseException:
                self.persistence_ok = False
                self.finalization_incomplete = True
                raise
            finally:
                if stream is not None: self.close_owned(stream, p)

    def deny_read(self, path, mode, flags, reason, *, source=None, cause=None):
        self.incomplete("READ_SCOPE_REFUSED")
        if len(self.meter["unexpected_read_denials"]) < 32:
            self.meter["unexpected_read_denials"].append({"path": self.evidence_text(str(path)),
                "mode": mode, "flags": flags, "phase": self.phase, "test_id": self.current,
                "reason": reason, "derived_source": None if source is None else self.evidence_text(str(source)),
                "cause": None if cause is None else self.exception_record(cause)})
        self.refuse("READ_SCOPE_REFUSED")

    def block_bound_cache_probe(self, path, mode, flags):
        """Deny cache bytes; CPython SourceLoader catches OSError and reads source.

        There is no permission to open this cache. A reversible current-tag
        mapping and a checked, SHA-pinned source are required for fallback.
        All unbound cache requests remain fatal policy failures.
        """
        try:
            source = Path(importlib.util.source_from_cache(str(path)))
        except ValueError as error:
            self.deny_read(path, mode, flags, "INVALID_CACHE_SOURCE_MAPPING", cause=error)
        pins = self.source_pins | self.runtime_code_pins
        if source.suffix.lower() != ".py" or source not in pins:
            self.deny_read(path, mode, flags, "CACHE_SOURCE_NOT_PINNED", source=source)
        expected = Path(importlib.util.cache_from_source(str(source), optimization=""))
        if path != expected:
            self.deny_read(path, mode, flags, "CACHE_TAG_OR_PATH_MISMATCH", source=source)
        # checked(path) inspects metadata only. The cache content is never read.
        try:
            checked(path)
            actual = sha(checked(source))
        except (OSError, Refusal) as error:
            self.deny_read(path, mode, flags, "CACHE_PATH_OR_SOURCE_IDENTITY", source=source, cause=error)
        if actual != pins[source]:
            self.deny_read(path, mode, flags, "CACHE_SOURCE_HASH_MISMATCH", source=source)
        key = str(path).lower()
        if key not in self.cache_probe_rows:
            require(len(self.cache_probe_rows) < 422, "CACHE_RECORD_LIMIT")
            row = {"cache_path": str(path), "source_path": str(source), "source_sha256": pins[source],
                "source_module": self.source_modules[source], "cache_tag": sys.implementation.cache_tag,
                "mode": mode, "flags": flags, "first_phase": self.phase, "phases_seen": [],
                "test_ids_seen": [], "attempts": 0, "decision": "DENIED_CACHE_USE_PINNED_SOURCE",
                "cache_bytes_read": 0, "fallback_exception": "PermissionError/EACCES"}
            self.cache_probe_rows[key] = row
            self.meter["expected_cache_probes"].append(row)
        row = self.cache_probe_rows[key]
        require(row["attempts"] < 8192, "CACHE_ATTEMPT_LIMIT")
        row["attempts"] += 1
        if self.phase not in row["phases_seen"]: row["phases_seen"].append(self.phase)
        if self.current is not None and self.current not in row["test_ids_seen"]:
            row["test_ids_seen"].append(self.current)
        raise PermissionError(errno.EACCES, "PINNED_SOURCE_CACHE_POLICY", str(path))

    def audit(self, event, args):
        if self.terminal and event in {"os.listdir", "os.scandir"}:
            self.terminal_read_gate(Path(args[0]), "directory_open")
        if self.terminal and event in {"import", "exec", "sqlite3.connect"}:
            self.refuse("TERMINAL_WORK_REFUSED")
        if event == "import":
            self.check_work()
            name = args[0]
            if name == "cryptography" or name.startswith("cryptography."):
                require(self.phase in {"D1A", "POSTCHECK"}, "A17_IMPORT_PRECONDITION")
            if any(name == p or name.startswith(p + ".") for p in FORBIDDEN_IMPORTS[1:]):
                self.refuse("PRODUCTION_MODULE_IMPORT_REFUSED")
        if event.startswith("socket."):
            self.meter["network"] += 1; self.refuse("NETWORK_REFUSED")
        if event in {"subprocess.Popen", "os.system", "os.startfile", "os.fork", "os.exec", "os.spawn"}:
            self.meter["subprocesses"] += 1; self.refuse("PROCESS_REFUSED")
        if event == "sqlite3.connect":
            if not self.connecting or self.phase not in {"A1", "D1A"} or args[0] != ":memory:":
                self.meter["file_SQLite"] += 1; self.refuse("SQLITE_SCOPE_REFUSED")
        if event in {"sqlite3.enable_load_extension", "sqlite3.load_extension"}:
            self.refuse("SQL_EXTENSION_REFUSED")
        if event == "exec":
            p = Path(args[0].co_filename)
            if p.is_absolute() and p.is_relative_to(PROJECT):
                require(p not in {PROJECT / "src/auth_lifecycle/online_state" / (name + ".py")
                    for name in ("services", "storage", "materials")}, "PRODUCTION_MODULE_EXEC_REFUSED")
                require(p in self.source_pins and sha(p) == self.source_pins[p], "EXEC_SOURCE_MISMATCH")
                self.exec_files.add(str(p))
            elif p.is_absolute() and p.is_relative_to(self.base_python):
                require(p in self.runtime_code_pins and sha(p) == self.runtime_code_pins[p],
                    "RUNTIME_EXEC_SOURCE_NOT_PINNED")
        if event == "open" and isinstance(args[0], (str, bytes, os.PathLike)):
            p = Path(os.fsdecode(args[0])).absolute()
            normalized = str(p).replace("\\", "/").lower()
            if p.suffix.lower() in {".db", ".sqlite", ".sqlite3", ".key", ".aes", ".pem", ".pfx"} or "/secure/" in normalized or "/.runtime/" in normalized:
                self.refuse("PRIVATE_FILE_REFUSED")
            mode, flags = args[1], args[2]
            writing = (isinstance(mode, str) and any(v in mode for v in "wax+")) or (
                isinstance(flags, int) and flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND))
            if writing:
                try: checked(p, output=True)
                except BaseException:
                    self.meter["outside_writes"] += 1; self.refuse("OUTPUT_SCOPE_REFUSED")
                require(p != APPROVAL, "APPROVAL_READ_ONLY")
                require(p == self.opening_output, "UNMETERED_OUTPUT_OPEN_REFUSED")
                if p.name in {x[0] for x in ALLOWED_D24_INPUTS}:
                    require(self.d24_active and self.current == "D24", "PATH_PROJECTION_REJECTED")
            elif self.terminal:
                self.terminal_read_gate(p, "content_open")
                checked(p, output=True)
            elif not self.terminal and time.perf_counter() >= self.work_deadline:
                self.refuse("WORK_DEADLINE_OR_STOP")
            elif p.suffix.lower() == ".pyc":
                self.block_bound_cache_probe(p, mode, flags)
            elif p not in self.read_files and p not in self.runtime_code_pins:
                if p.is_relative_to(OUTPUT):
                    checked(p, output=True)
                else:
                    self.deny_read(p, mode, flags, "READ_PATH_NOT_EXPLICITLY_PINNED")
        if event == "os.mkdir":
            require(Path(args[0]).absolute() == RUN, "DIRECTORY_SCOPE_REFUSED")
            checked(RUN, output=True)
        if event in {"os.remove", "os.rmdir", "os.rename", "os.chmod", "os.link", "os.symlink"}:
            self.refuse("FILESYSTEM_MUTATION_REFUSED")

    def preflight(self):
        self.check_work()
        self.build_verified = False
        require(Path(sys.executable) == Path(self.build["python"]["path"])
            and sha(sys.executable) == self.build["python"]["sha256"]
            and sys.version_info[:3] == (3, 12, 14) and sys.flags.isolated == 1
            and sys.dont_write_bytecode and sys.flags.utf8_mode == 1 and sys.flags.optimize == 0
            and Path.cwd() == PROJECT, "INTERPRETER_OR_CWD_MISMATCH")
        if self.phase == "PREFLIGHT":
            require(not any(name == "auth_lifecycle" or name.startswith("auth_lifecycle.")
                or name.startswith("r33_cases_") for name in sys.modules), "PRELOADED_PROJECT_MODULE_REFUSED")
        pins = self.build["engineering_source"] + self.build["regression_extension_sources"] + self.build["runtime_files"]
        require(all(sha(checked(x["path"])) == x["sha256"] for x in pins), "BUILD_IDENTITY_MISMATCH")
        require(hashlib.sha256(canonical(self.build["engineering_source"])).hexdigest()
            == self.build["source_digest"], "BUILD_DIGEST_MISMATCH")
        require(all(sha(checked(path)) == digest for path, digest in self.runtime_code_pins.items()),
            "RUNTIME_READ_CATALOGUE_CHANGED")
        for row in self.history["protected_files"] + self.history["additional_records"]:
            self.check_work()
            require(sha(checked(row["path"], historical_read=True)) == row["sha256"], "HISTORY_CHANGED")
        specs = self.manifest["tests"]
        require(len(specs) == len({x["test_id"] for x in specs}) == 68, "TEST_COUNT_MISMATCH")
        for phase, count in (("A1", 20), ("H1", 24), ("D1A", 24)):
            require(len([x for x in specs if x["test_source"] == phase]) == count, "TEST_GROUP_MISMATCH")
        for row in specs:
            require(sha(row["test_file"]) == row["source_hash"]
                and sha(row["original_plan"]["path"]) == row["original_plan"]["sha256"], "TEST_SOURCE_CHANGED")
            for f in row["fixture_hash"]:
                require(sha(f["path"]) == f["sha256"], "FIXTURE_CHANGED")
                if Path(f["path"]).name == "CANARIES.json":
                    self.secrets.update(load(f["path"])["values"].values())
        for name, path, digest in ALLOWED_D24_INPUTS:
            require(path.name == name and sha(checked(path)) == digest, "D24_ORIGINAL_CHANGED")
        self.build_verified = True

    def origin_observation(self, name, module):
        # Read dictionaries only: module-level __getattr__ may perform imports.
        metadata = module_dictionary(module)
        spec = metadata.get("__spec__")
        fields = vars(spec) if type(spec) is importlib.machinery.ModuleSpec else {}
        loader = fields.get("loader")
        expected = self.expected_names.get(name)
        pins = self.source_pins | self.runtime_code_pins
        def field(value):
            if type(value) is str: return self.evidence_text(value)
            if value is None or type(value) is bool: return value
            if type(value) is int and value.bit_length() <= 64: return value
            return "UNKNOWN"
        require(type(name) is str, "ORIGIN_UNRESOLVED")
        name = self.evidence_text(name, 256)
        loader_type = loader if type(loader) is type else type(loader)
        loader_label = (type.__dict__["__module__"].__get__(loader_type, type(loader_type)) + "."
            + type.__dict__["__qualname__"].__get__(loader_type, type(loader_type)))
        loader_data = vars(loader) if type(loader) in (importlib.machinery.SourceFileLoader,
            importlib.machinery.ExtensionFileLoader) else {}
        return {"module_name": name, "module_type": self.evidence_text(builtin_type_field(module, "__name__"), 256),
            "module_type_module": self.evidence_text(builtin_type_field(module, "__module__"), 256), "object_id": id(module),
            "declared_name": field(metadata.get("__name__", "UNKNOWN")),
            "source_path": field(metadata.get("__file__", "UNKNOWN")),
            "origin": field(fields.get("origin", "UNKNOWN")), "spec_name": field(fields.get("name", "UNKNOWN")),
            "loader": "UNKNOWN" if loader is None else self.evidence_text(loader_label, 512),
            "loader_name": field(loader_data.get("name", "UNKNOWN")),
            "resolved_path": "UNKNOWN", "source_digest": "UNKNOWN",
            "expected_origin": str(expected) if expected is not None else "UNKNOWN",
            "expected_digest": pins.get(expected, "UNKNOWN"), "classification": "OBSERVED_UNVALIDATED",
            "evidence": []}

    def verify_file_origin(self, row, module, expected):
        require(type(module) is type(sys), "ORIGIN_UNRESOLVED")
        require(expected is not None, "UNAUTHORIZED_ORIGIN")
        path, origin = row["source_path"], row["origin"]
        require(isinstance(path, str) and path != "UNKNOWN"
            and isinstance(origin, str) and origin != "UNKNOWN", "ORIGIN_UNRESOLVED")
        require(Path(path) == expected and Path(origin) == expected, "DEPENDENCY_ORIGIN_MISMATCH")
        require(row["declared_name"] == row["spec_name"] == row["module_name"],
            "DEPENDENCY_ORIGIN_MISMATCH")
        loader = vars(vars(module)["__spec__"])["loader"]
        allowed_loader = importlib.machinery.ExtensionFileLoader if expected.suffix == ".pyd" else importlib.machinery.SourceFileLoader
        require(type(loader) is allowed_loader and loader.name == row["module_name"]
            and Path(loader.path) == expected, "DEPENDENCY_ORIGIN_MISMATCH")
        path = checked(path)
        row["resolved_path"] = str(path.resolve(strict=True))
        require(path.samefile(expected) and Path(origin).samefile(expected), "DEPENDENCY_ORIGIN_MISMATCH")
        stat = path.stat()
        row["file_identity"] = {"device": stat.st_dev, "inode": stat.st_ino,
            "size": stat.st_size, "mtime_ns": stat.st_mtime_ns}
        pins = self.source_pins | self.runtime_code_pins
        require(expected in pins, "UNAUTHORIZED_ORIGIN")
        row["source_digest"] = sha(path)
        require(row["source_digest"] == pins[expected], "DEPENDENCY_CHANGED")
        after = path.stat()
        require((after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns)
            == (stat.st_dev, stat.st_ino, stat.st_size, stat.st_mtime_ns), "DEPENDENCY_ORIGIN_MISMATCH")
        row["classification"] = "TRUSTED_FROZEN_SOURCE" if expected in self.source_pins else "TRUSTED_RUNTIME_SOURCE"
        row["evidence"].append("exact name/file/spec/loader/path identity and locked SHA256")

    def verify_dynamic_origin(self, row, module):
        name = row["module_name"]
        require(type(module) is type(sys) and name.startswith(NATIVE_PARENT + "."), "ORIGIN_UNRESOLVED")
        parent = sys.modules.get(NATIVE_PARENT)
        parent_row = self.origin_observation(NATIVE_PARENT, parent)
        expected = self.base_python / "Lib/site-packages/cryptography/hazmat/bindings/_rust.pyd"
        require(self.runtime_code_pins.get(expected) == NATIVE_PARENT_SHA256, "UNAUTHORIZED_ORIGIN")
        self.verify_file_origin(parent_row, parent, expected)
        target = parent
        chain = []
        for part in name[len(NATIVE_PARENT) + 1:].split("."):
            require(type(target) is type(sys), "ORIGIN_UNRESOLVED")
            target = vars(target).get(part)
            chain.append(part)
        require(target is module and sys.modules.get(name) is module, "DEPENDENCY_ORIGIN_MISMATCH")
        # A child may omit file/spec metadata, or point at this exact binary.
        for field in ("source_path", "origin"):
            value = row[field]
            require(value in (None, "UNKNOWN") or (isinstance(value, str) and Path(value) == expected),
                "DEPENDENCY_ORIGIN_MISMATCH")
        spec = vars(module).get("__spec__")
        require(spec is None or (type(spec) is importlib.machinery.ModuleSpec
            and spec.name == name and spec.loader is None), "DEPENDENCY_ORIGIN_MISMATCH")
        row.update({"classification": "EXPECTED_DYNAMIC_SOURCE", "resolved_path": parent_row["resolved_path"],
            "source_digest": parent_row["source_digest"], "expected_origin": str(expected),
            "expected_digest": NATIVE_PARENT_SHA256, "parent_file_identity": parent_row["file_identity"],
            "parent_module": NATIVE_PARENT, "parent_attribute_chain": chain,
            "parent_object_identity_confirmed": True})
        row["evidence"].append("pinned ExtensionFileLoader parent and identical child reached through module dictionaries")

    def classify_origin(self, row, module):
        if type(module) is not type(sys):
            self.verify_wrapper_origin(row, module)
            return
        name, origin = row["module_name"], row["origin"]
        require(sys.modules.get(name) is module, "DEPENDENCY_ORIGIN_MISMATCH")
        spec = vars(module).get("__spec__")
        if name == "__main__":
            expected = Path(__file__).absolute()
            require(spec is None and Path(row["source_path"]) == expected
                and sys.modules.get(name) is module, "DEPENDENCY_ORIGIN_MISMATCH")
            path = checked(expected)
            row.update({"resolved_path": str(path.resolve(strict=True)), "source_digest": sha(path)})
            require(row["source_digest"] == self.command["runner_sha256"], "DEPENDENCY_CHANGED")
            row["classification"] = "TRUSTED_FROZEN_SOURCE"
            row["evidence"].append("direct-script __main__;command-pinned file SHA and sys.modules object identity")
        elif origin in {"built-in", "frozen"}:
            require(type(spec) is importlib.machinery.ModuleSpec, "ORIGIN_UNRESOLVED")
            expected_loader = importlib.machinery.BuiltinImporter if origin == "built-in" else importlib.machinery.FrozenImporter
            registered = _imp.is_builtin(spec.name) if origin == "built-in" else _imp.is_frozen(spec.name)
            require(spec.loader is expected_loader and registered and sys.modules.get(spec.name) is module,
                "DEPENDENCY_ORIGIN_MISMATCH")
            row["classification"] = "TRUSTED_BUILTIN" if origin == "built-in" else "TRUSTED_FROZEN_RUNTIME"
            row["evidence"].append("interpreter registry,canonical spec loader and registered object;runtime24 pinned in preflight")
        elif name.startswith(NATIVE_PARENT + "."):
            self.verify_dynamic_origin(row, module)
        else:
            self.verify_file_origin(row, module, self.expected_names.get(name))

    def verify_wrapper_origin(self, row, module):
        name = row["module_name"]
        require(name in WRAPPER_NAMES, "ORIGIN_UNRESOLVED")
        utils = sys.modules.get("cryptography.utils")
        require(type(utils) is type(sys), "ORIGIN_UNRESOLVED")
        utils_path = self.base_python / "Lib/site-packages/cryptography/utils.py"
        require(self.runtime_code_pins.get(utils_path) == CRYPTO_UTILS_SHA256, "UNAUTHORIZED_ORIGIN")

    def connect_memory(self):
        self.check_work()
        require(self.phase in {"A1", "D1A"}, "SQLITE_PHASE_REFUSED")
        cap = 32 if self.phase == "A1" else 8
        require(self.meter["memory_connections"][self.phase] < cap
            and sum(self.meter["memory_connections"].values()) < 40
            and self.meter["memory_active"] < 32, "SQLITE_CONNECTION_BUDGET")
        self.meter["memory_connections"][self.phase] += 1
        self.connecting = True
        try:
            connection = sqlite3.connect(":memory:", isolation_level=None, check_same_thread=False,
                factory=AuditedConnection)
        finally:
            self.connecting = False
        connection.owner, connection.owner_closed = self, False
        self.connections.append(connection)
        self.meter["memory_success"] += 1; self.meter["memory_active"] += 1
        self.meter["memory_peak"] = max(self.meter["memory_peak"], self.meter["memory_active"])
        connection.row_factory = sqlite3.Row
        def policy(action, one, two, database, trigger):
            if action in {sqlite3.SQLITE_ATTACH, sqlite3.SQLITE_DETACH} or (
                    action == sqlite3.SQLITE_FUNCTION and str(two).lower() == "load_extension"):
                return sqlite3.SQLITE_DENY
            return sqlite3.SQLITE_OK
        connection.set_authorizer(policy)
        return connection

    def sql(self, text):
        self.check_work()
        require(self.phase in {"A1", "D1A"} and self.meter["execute_calls"] < 376, "SQL_EXECUTE_BUDGET")
        self.meter["execute_calls"] += 1
        kind = str(text).strip().split()[0].upper()
        kind = self.evidence_text(kind, 128)
        self.meter["SQL_kinds"][kind] = self.meter["SQL_kinds"].get(kind, 0) + 1

    def close_connections(self):
        first = None
        for connection in self.connections:
            if not connection.owner_closed:
                if time.perf_counter() >= self.deadline:
                    self.incomplete("CONNECTION_CLEANUP_DEADLINE")
                    raise Refusal("CONNECTION_CLEANUP_DEADLINE")
                try: connection.close()
                except BaseException as error:
                    self.failure("CLEANUP_CONNECTION", error)
                    if first is None: first = error
        if first is not None: raise first


class AuditedConnection(sqlite3.Connection):
    def execute(self, sql, parameters=()):
        self.owner.sql(sql)
        return super().execute(sql, parameters)

    def executemany(self, *args, **kwargs):
        raise Refusal("UNEXPECTED_SQL_API")

    executescript = executemany
    commit = executemany

    def cursor(self, *args, **kwargs):
        kwargs["factory"] = AuditedCursor
        return super().cursor(*args, **kwargs)

    def rollback(self):
        require(self.owner.meter["rollback_calls"] < 1, "ROLLBACK_BUDGET")
        self.owner.meter["rollback_calls"] += 1
        return super().rollback()

    def close(self):
        if not self.owner_closed:
            super().close(); self.owner_closed = True
            self.owner.meter["memory_active"] -= 1

class Sink(io.TextIOBase):
    def __init__(self, state, path):
        self.state, self.path = state, checked(path, output=True)
        self.stream = None
        self.attempts = 0
        state.reserve_bytes(0, self.path)
        self.stream = state.open_owned(self.path)
        state.streams.append(self)

    def write(self, text):
        require(type(text) is str, "LOG_TEXT_TYPE")
        try:
            require(len(text) <= 65536, "LOG_BYTE_LIMIT")
            clean = self.state.redact(text)
            # At most 4 bytes per Unicode code point, followed by the exact cap.
            require(len(clean) <= 65536, "LOG_BYTE_LIMIT")
            data = clean.encode("utf-8")
            with self.state.io_lock:
                require(self.attempts < MAX_STREAM_WRITES, "LOG_WRITE_ATTEMPT_LIMIT")
                self.attempts += 1
                self.state.reserve_bytes(len(data), self.path)
                self.state.io_write(self.stream, self.path, data)
        except BaseException:
            self.state.persistence_ok = False
            self.state.stop_requested = True
            raise
        return len(text)

    def flush(self):
        # io_write already checks flush/fsync. FileIO has zero pending payload.
        return None

    def close(self):
        if not self.closed:
            try: self.state.close_owned(self.stream, self.path)
            finally: super().close()


class ProjectionFile:
    def __init__(self, projection, name):
        require(name in projection.allowed, "PATH_PROJECTION_REJECTED")
        self.projection, self.name = projection, name

    def __str__(self):
        return str(LOGICAL / self.name)

    def write_bytes(self, data):
        p = self.projection
        require(p.state.d24_active and type(data) is bytes
            and hashlib.sha256(data).hexdigest() == p.allowed[self.name][1], "PATH_PROJECTION_REJECTED")
        p.state.write(RUN / self.name, data, raw=True)
        p.written.add(self.name)
        return len(data)


class ProjectionRead:
    def __init__(self, projection, name):
        self.projection, self.name = projection, name

    def _path(self):
        p = self.projection
        require(p.state.d24_active and p.state.current == "D24" and self.name in p.written,
            "PATH_PROJECTION_REJECTED")
        target = checked(RUN / self.name, output=True)
        require(sha(target) == p.allowed[self.name][1], "PATH_PROJECTION_REJECTED")
        return target

    @property
    def parents(self): return self._path().parents
    def exists(self): return self._path().exists()
    def is_symlink(self): return self._path().is_symlink()
    def lstat(self): return self._path().lstat()
    def stat(self): return self._path().stat()
    def read_bytes(self): return self._path().read_bytes()


class D24Projection:
    def __init__(self, state, module):
        self.state, self.module = state, module
        self.allowed = {name: (path, digest) for name, path, digest in ALLOWED_D24_INPUTS}
        self.written = set()

    def __truediv__(self, name):
        return ProjectionFile(self, name)

    def __enter__(self):
        require(self.state.phase == "D1A" and self.state.current == "D24", "PATH_PROJECTION_REJECTED")
        for path, digest in self.allowed.values():
            require(sha(checked(path)) == digest, "PATH_PROJECTION_REJECTED")
        self.old_path, self.old_output = self.module.r.Path, self.module.CONFIG["output"]
        def provider(value):
            path = self.old_path(value)
            require(path.parent == LOGICAL and path.name in self.allowed, "PATH_PROJECTION_REJECTED")
            return ProjectionRead(self, path.name)
        self.module.r.Path = provider
        self.module.CONFIG["output"] = self
        self.state.d24_active = True
        return self

    def __exit__(self, kind, value, trace):
        self.module.r.Path = self.old_path
        self.module.CONFIG["output"] = self.old_output
        self.state.d24_active = False
        self.state.projections.append({"test_id": "D24", "written_names": sorted(self.written),
            "context_restored": self.module.r.Path is self.old_path and self.module.CONFIG["output"] is self.old_output,
            "inputs": [{"original": str(path), "sha256": digest, "target": str(RUN / name)}
                for name, (path, digest) in self.allowed.items()]})
        if kind is None:
            require(self.written == set(self.allowed), "PATH_PROJECTION_INCOMPLETE")


class Results(unittest.TextTestResult):
    def __init__(self, *args, state, **kwargs):
        super().__init__(*args, **kwargs); self.state = state

    def startTest(self, test):
        self.started = time.perf_counter_ns()
        tid = test._testMethodName.split("_")[1]
        self.state.begin_case(tid)
        super().startTest(test)

    def _exc_info_to_string(self, error, test):
        descriptor = self.state.exception_record(error[1])
        raw = bounded_canonical(descriptor, MAX_EXCEPTION_JSON_BYTES).decode("ascii")
        try: text = self.state.evidence_text(raw, MAX_TEXT_JSON_BYTES)
        except Refusal:
            text = "EVIDENCE_CONTRACT_UNSATISFIED: bounded exception descriptor exceeds test text field"
        tid = test._testMethodName.split("_")[1]
        if tid in self.state.rows:
            self.state.rows[tid]["detail"] = text
            self.state.rows[tid]["diagnostic_status"] = "INCOMPLETE_ORIGINAL_FULL_TEXT_NOT_PRESERVED"
        return text

    def record(self, test, status, error=None):
        tid = test._testMethodName.split("_")[1]
        return self.state.observe_result(tid, status, time.perf_counter_ns() - self.started, error)

        self.record(test, "PASS")
        super().addSuccess(test)

        super().addFailure(test, error)

        super().addError(test, error)

        self.state.rows[test._testMethodName.split("_")[1]]["detail"] = self.state.evidence_text(reason, MAX_TEXT_JSON_BYTES)
        super().addSkip(test, reason)


class OrderedSuite(unittest.TestSuite):
    def __init__(self, cases, state, module):
        super().__init__(cases); self.state, self.module = state, module

    def run(self, result, debug=False):
        for test in self:
            self.state.check_work()
            if result.shouldStop: break
            tid = test._testMethodName.split("_")[1]
            self.state.current = tid
            escaped = False
            try:
                if tid == "D24":
                    with D24Projection(self.state, self.module): test(result)
                else:
                    test(result)
            except BaseException:
                escaped = True
                self.state.begin_terminal()
                raise
            finally:
                try:
                    self.state.end_case(tid, self.module, test, escaped)
                except BaseException as error:
                    self.state.incomplete("CASE_EVIDENCE_FINALIZATION_FAILED")
                    self.state.failure("CASE_EVIDENCE_FINALIZATION", error)
                    if not escaped: raise
            if result.skipped: result.stop()
        self.state.current = None
        return result


def load_group(state, phase, classname):
    state.check_work()
    require(state.phase == phase, "PHASE_ORDER_MISMATCH")
    if phase in {"A1", "H1"}:
        require(not any(name == p or name.startswith(p + ".") for name in sys.modules
                        for p in FORBIDDEN_IMPORTS), "A17_IMPORT_PRECONDITION")
    rows = [x for x in state.manifest["tests"] if x["test_source"] == phase]
    path = Path(rows[0]["test_file"]); alias = "r33_cases_" + phase.lower()
    require(alias not in sys.modules, "TEST_MODULE_ALREADY_LOADED")
    spec = importlib.util.spec_from_file_location(alias, path)
    module = importlib.util.module_from_spec(spec); sys.modules[alias] = module
    source = path.read_bytes()
    require(hashlib.sha256(source).hexdigest() == rows[0]["source_hash"], "TEST_SOURCE_CHANGED")
    # Compile the pinned test bytes; no historical test bytecode is substituted.
    exec(compile(source, str(path), "exec"), module.__dict__)
    cls = getattr(module, classname)
    names = [x["test_callable"].split(".")[1] for x in rows]
    require(set(name for name in vars(cls) if name.startswith("test_")) == set(names)
        and len(names) == len(set(names)), "TEST_DISCOVERY_MISMATCH")
    plan = load(rows[0]["original_plan"]["path"])
    fixture = Path(rows[0]["fixture_hash"][0]["path"]).parent
    if phase == "A1": module.configure(plan, fixture, state.connect_memory)
    elif phase == "H1": module.configure(plan, fixture)
    else: module.configure(plan, fixture, state.connect_memory, RUN)
    state.origins_snapshot(phase + "_loaded")
    return module, OrderedSuite([cls(name) for name in names], state, module)


def peak_memory():
    if os.name != "nt": return {"status": "NOT_MEASURED"}
    try:
        from ctypes import wintypes as W
        class Memory(ctypes.Structure):
            _fields_ = [("cb", W.DWORD), ("PageFaultCount", W.DWORD)] + [(n, ctypes.c_size_t) for n in
                ("PeakWorkingSetSize", "WorkingSetSize", "QuotaPeakPagedPoolUsage", "QuotaPagedPoolUsage",
                 "QuotaPeakNonPagedPoolUsage", "QuotaNonPagedPoolUsage", "PagefileUsage", "PeakPagefileUsage", "PrivateUsage")]
        kernel = ctypes.WinDLL("kernel32", use_last_error=True)
        kernel.GetCurrentProcess.restype = W.HANDLE
        kernel.K32GetProcessMemoryInfo.argtypes = [W.HANDLE, ctypes.POINTER(Memory), W.DWORD]
        value = Memory(); value.cb = ctypes.sizeof(value)
        require(kernel.K32GetProcessMemoryInfo(kernel.GetCurrentProcess(), ctypes.byref(value), value.cb), "MEMORY_QUERY_FAILED")
        return {"status": "MEASURED", "peak_working_set_bytes": value.PeakWorkingSetSize,
            "basis": "current regression interpreter only"}
    except Exception:
        return {"status": "NOT_MEASURED", "reason": "MEMORY_QUERY_UNAVAILABLE"}

        "output_bytes_at_resource_audit": state.sizes(), "byte_accounting": state.byte_ledger,
        "case_partition": state.case_partition(), "detail_checkpoint_refs": state.detail_refs,
        "terminal_read_audit": state.read_audit,
        "evidence_complete": state.evidence_complete, "evidence_issues": sorted(state.evidence_issues),
        "finalization_incomplete": state.finalization_incomplete,
        "terminal_write_attempts": state.terminal_attempts,
        "accounting_snapshot": "at ledger serialization;later writes retained in memory/receipt when available"}
    ledger_written = state.publish("RESOURCE_LEDGER.json", ledger)
    eligible = (assertion_ok and case_evidence_ok and provenance_ok and resource_ok and integrity and primary is None
        and not state.errors and not state.meter["denials"] and state.persistence_ok
        and origins_written and ledger_written and state.evidence_complete
        and not state.finalization_incomplete)
    if not state.safe_checkpoint("FINALIZATION_READY", {"counts": counts, "eligible": eligible,
        "errors": state.errors, "artifacts": state.artifacts}): eligible = False
    final = {"schema": "r38-current-build-regression68/1", "suite": "regression68", "counts": counts,
        "cases": rows, "details": state.details, "passed": eligible,
        "case_partition": state.case_partition(), "case_evidence_complete": case_evidence_ok,
        "TEST_ASSERTIONS": {"counts": counts, "all_68_passed": assertion_ok},
        "DEPENDENCY_PROVENANCE": {"verified": provenance_ok, "failures": state.origin_failures},
        "RESOURCE_AND_INTEGRITY": {"resource_ok": resource_ok, "integrity_ok": integrity},
        "FINAL_VERDICT": {"eligible_for_acceptance": eligible, "commit_receipt_required": True,
            "rule": "TEST_RUNS.json must bind all artifact hashes and exit must be0;RESULTS alone is a candidate"},
        "argv": sys.orig_argv, "python": sys.version, "started_utc": state.started_utc,
        "runtime_versions": {"python": sys.version, "sqlite": sqlite3.sqlite_version,
            "cryptography": vars(sys.modules["cryptography"]).get("__version__", "UNKNOWN")
                if "cryptography" in sys.modules else "UNKNOWN"},
        "finished_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
        "source_digest": state.build["source_digest"], "engineering_source": state.build["engineering_source"],
        "runner_sha256": state.command["runner_sha256"], "fixture_hashes": state.manifest["fixture_files"],
        "command_lock_sha256": state.command_digest,

        "executed_order": list(state.rows), "errors": list(state.errors),
        "resources": ledger, "D1B1_OFFLINE_CONTRACT_COMPLETE": False,
        "D1B2_P1A_P1B_P2_S3B2_S5": False}
    results_written = state.publish("RESULTS.json", final)
    eligible = eligible and results_written
    if not state.safe_checkpoint("FINALIZED" if results_written and origins_written and ledger_written else "RECOVERY_REQUIRED",
        {"eligible": eligible, "commit_pending": True, "artifacts": state.artifacts,
         "errors": state.errors, "counts": counts}): eligible = False
    return eligible, counts


def main(argv=None):
    global ENTRY_STATE
    started = time.perf_counter()
    started_utc = datetime.datetime.now(datetime.timezone.utc).isoformat()
    parser = Parser(add_help=False)
    parser.add_argument("--suite", choices=["regression68"], required=True)
    parser.add_argument("--output-root", required=True)
    parser.add_argument("--approval-ref", required=True)
    try:
        args = parser.parse_args(argv)
        command, approval = controls(args)
        state = State(command, started, started_utc)
        ENTRY_STATE = state
    except Exception:
        print("REGRESSION_ADMISSION_REFUSED", file=sys.stderr)
        return 2
    sys.addaudithook(state.audit)
    state.check_work()
    require(all(name in sys.builtin_module_names for name in ("binascii", "zlib")),
        "CODEC_NOT_FROZEN_BUILTIN")
    import binascii
    import zlib
    state.codec = (binascii, zlib)
    # This exclusive claim is the only invocation slot. It is never reset.
    state.write(OUTPUT / "SUITE_INVOCATION_01.json", {"suite": "regression68", "argv": sys.orig_argv,
        "started_utc": state.started_utc, "approval_sha256": sha(APPROVAL), "max_runs": 1})
    checked(RUN, output=True).mkdir()
    primary = None
    timer = None
    original_start = threading.Thread.start
    def thread_start(thread, *args, **kwargs):
        state.check_work()
        require(state.meter["threads_created"] < 12 and threading.active_count() < 6, "THREAD_BUDGET")
        state.meter["threads_created"] += 1
        result = original_start(thread, *args, **kwargs)
        state.meter["threads_peak"] = max(state.meter["threads_peak"], threading.active_count())
        return result
    with Sink(state, OUTPUT / "COMMAND_stdout.log") as stdout, Sink(state, OUTPUT / "COMMAND_stderr.log") as stderr:
        with contextlib.redirect_stdout(stdout), contextlib.redirect_stderr(stderr):
            try:
                state.preflight()
                state.write(RUN / "RUN_START.json", {"argv": sys.orig_argv, "python": sys.version,
                    "cwd": str(Path.cwd()), "started_utc": state.started_utc,
                    "source_digest": state.build["source_digest"], "runner_sha256": sha(__file__),
                    "command_lock_sha256": sha(RECORD / "R35_REGRESSION_COMMAND_LOCK.json"),
                    "approval_sha256": sha(APPROVAL),
                    "test_ids": [x["test_id"] for x in state.manifest["tests"]]})
                state.checkpoint("PRECHECK", {"source_digest": state.build["source_digest"],
                    "argv": sys.orig_argv, "runner_sha256": command["runner_sha256"],
                    "command_lock_sha256": sha(RECORD / "R35_REGRESSION_COMMAND_LOCK.json")})
                state.checkpoint("RUN_STARTED", {"run_id": RUN_ID, "max_suite_runs": 1,
                    "automatic_reruns": 0, "resource_lock": state.resources})
                state.origins_snapshot("PRE_TEST_DEPENDENCY_SET")
                threading.Thread.start = thread_start
                timer = threading.Timer(max(.01, state.work_deadline - time.perf_counter()), _thread.interrupt_main)
                timer.daemon = True; timer.start()
                sys.path.insert(0, str(PROJECT / "src"))
                with Sink(state, RUN / "stdout.log") as log:
                    for phase, cls in PHASES:
                        state.phase = phase
                        state.checkpoint("TESTS_RUNNING", {"phase": phase})
                        module, suite = load_group(state, phase, cls)
                        result = unittest.TextTestRunner(stream=log, verbosity=2, failfast=True,
                            resultclass=lambda *a, **kw: Results(*a, state=state, **kw)).run(suite)
                        state.checkpoint("PHASE_ASSERTIONS_FINISHED", {"phase": phase,
                            "counts": state.assertion_counts(), "meter": state.meter,
                            "details": module.DETAILS, "projection": state.projections})
                        if all(x != "NOT_RUN" for x in state.case_lifecycle.values()) or not result.wasSuccessful() or result.skipped:
                            state.finish_assertions()
                        state.origins_snapshot(phase + "_finished")
                        state.close_connections()
                        if not result.wasSuccessful() or result.skipped: break
                state.finish_assertions()
            except BaseException as error:
                primary = error
                state.begin_terminal()
                state.safe_checkpoint("FAILED", state.failure("EXECUTION", error))
            finally:
                if timer is not None:
                    timer.cancel()
                    try:
                        timer.join(max(0, min(1.0, state.deadline - time.perf_counter())))
                        require(not timer.is_alive(), "WATCHDOG_REAP_INCOMPLETE")
                    except BaseException as error:
                        if primary is None: primary = error
                        state.safe_checkpoint("FAILED", state.failure("CLEANUP_WATCHDOG", error))
                threading.Thread.start = original_start
                try: state.close_connections()
                except BaseException as error:
                    if primary is None: primary = error
                    state.safe_checkpoint("FAILED", state.failure("CLEANUP_CONNECTIONS", error))
            if not state.terminal:
                try: collect_work_audits(state)
                except BaseException as error:
                    if primary is None: primary = error
                    state.begin_terminal()
                    state.safe_checkpoint("FAILED", state.failure("WORK_AUDIT", error))
            passed, counts = finalize_evidence(state, primary)
            print(json.dumps({"counts": counts, "eligible_pending_commit": passed, "results": str(RUN / "RESULTS.json")}))
    # All redirected streams are closed/fsynced before this logical commit.
    artifacts = dict(state.artifacts)
    for path in state.output_files():
        artifacts[path.relative_to(OUTPUT).as_posix()] = {"path": str(path), "sha256": sha(path)}
    receipt_ok = state.publish("TEST_RUNS.json", {"schema": "r38-run-commit/1", "status": "FINALIZED" if passed else "FAILED",
        "FULL_CONTRACT_ACCEPTED": passed, "artifacts": artifacts, "journal_head": state.journal_head,
        "journal_events": state.journal_sequence, "errors": list(state.errors),
        "runs": [{"path": str(RUN / "RESULTS.json"), "sha256": state.artifacts.get("RESULTS.json", {}).get("sha256", "UNKNOWN"),
            "counts": counts, "passed": passed}], "native_exit_required": 0 if passed else 1,
        "byte_accounting_before_receipt": state.byte_ledger,
        "case_partition": state.case_partition(), "terminal_read_audit": state.read_audit,
        "native_exit_observed": "UNKNOWN",
        "evidence_complete": state.evidence_complete, "evidence_issues": sorted(state.evidence_issues),
        "FINALIZATION_INCOMPLETE": state.finalization_incomplete})
    try: state.close_journal()
    except BaseException as error:
        state.failure("PERSIST_JOURNAL_CLOSE", error)
        passed = False
    timely = time.perf_counter() < state.terminal_deadline
    state.close_read_handles()
    handles_closed = all(stream.closed for stream in state.owned_handles.values()) and all(
        item["stream"] is None or item["stream"].closed for item in state.read_handles)
    return 0 if (passed and receipt_ok and timely and handles_closed and state.persistence_ok
        and not state.finalization_incomplete) else 1


def entry():
    try:
        return main()
    except BaseException as error:
        if ENTRY_STATE is not None:
            state = ENTRY_STATE
            state.begin_terminal()
            state.finalization_incomplete = True
            row = state.failure("HARNESS_FINALIZATION_ERROR", error)
            try:
                if metadata(RUN, "is_dir") and not state.journal_closed:
                    state.safe_checkpoint("RECOVERY_REQUIRED", {"error": row,
                        "counts": state.assertion_counts(), "recorded_test_ids": list(state.rows),
                        "run_incomplete": True})
                if "TEST_RUNS.json" not in state.write_attempted_paths:
                    state.publish("TEST_RUNS.json", {"schema": "r38-run-commit/1", "status": "FAILED",
                        "FULL_CONTRACT_ACCEPTED": False, "FINALIZATION_INCOMPLETE": True,
                        "counts": state.assertion_counts(), "error": row,
                        "case_partition": state.case_partition(), "native_exit_observed": "UNKNOWN",
                        "journal_head": state.journal_head, "journal_events": state.journal_sequence,
                        "native_exit_required": 1})
            except BaseException as evidence_error:
                state.failure("PERSIST_ENTRY_FALLBACK", evidence_error)
            finally:
                for sink in state.streams:
                    try: sink.close()
                    except BaseException: state.persistence_ok = False
                try: state.close_journal()
                except BaseException: state.persistence_ok = False
                try: state.close_read_handles()
                except BaseException: state.persistence_ok = False
                for path, stream in state.owned_handles.items():
                    try: state.close_owned(stream, path)
                    except BaseException: state.persistence_ok = False
        return 1


if __name__ == "__main__":
    try:
        exit_code = entry()
    except BaseException:
        # Last-resort recorder/allocation failure: no recursive formatter or
        # unmanaged traceback. Existing evidence prefix remains incomplete.
        exit_code = 1
    raise SystemExit(exit_code)
