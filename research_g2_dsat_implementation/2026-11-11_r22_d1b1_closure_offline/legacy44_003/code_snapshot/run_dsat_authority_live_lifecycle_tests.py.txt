def load(path): return json.loads(Path(path).read_text(encoding="utf-8-sig"))
def sha(path): return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def target(path):
    p = Path(path).absolute()
    if not p.is_relative_to(OUTPUT): raise RuntimeError("OUTPUT_PATH_DENIED")
    for part in (p, *p.parents):
        if part.exists() and (part.is_symlink() or getattr(part.lstat(), "st_file_attributes", 0) & 0x400):
            raise RuntimeError("OUTPUT_REPARSE_DENIED")
    return p


def write(path, value):
    target(path).write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def redact(value):
    for secret in sorted(SECRETS, key=len, reverse=True): value = value.replace(secret, "[REDACTED_TEST_INPUT]")
    return value


def violation(kind):
    AUDIT[kind] += 1
    AUDIT["unexpected_violations"].append(kind)
    raise RuntimeError("R21_ISOLATION_DENIED")


def audit(event, args):
    if event == "sqlite3.connect" and PHASE == "worker": violation("private_outside_access")
    if event.startswith("socket."): violation("socket_attempts")
    if event in {"os.system", "os.startfile", "os.fork", "os.exec"}: violation("forbidden_process_attempts")
    if event == "subprocess.Popen":
        allowed = PHASE == "new54" and RUN is not None
        command = args[1]
        runner = PROJECT / "scripts/run_dsat_authority_live_lifecycle_tests.py"
        candidates = [] if RUN is None else [
            [sys.executable, "-B", str(runner), "--worker", "--input", str(p)] for p in RUN.glob("worker_*.json")]
        allowed = allowed and any(command == candidate or command == subprocess.list2cmdline(candidate) for candidate in candidates)
        allowed = allowed and AUDIT["child_launch_attempts"] < 8
        if not allowed: violation("forbidden_process_attempts")
        AUDIT["child_launch_attempts"] += 1
    if event in {"sqlite3.enable_load_extension", "sqlite3.load_extension"}: violation("private_outside_access")
    if event == "open" and isinstance(args[0], (str, bytes, os.PathLike)):
        p = Path(os.fsdecode(args[0])).absolute()
        if p.is_relative_to(DEPLOY): violation("deployment_access")
        is_private = p.suffix.lower() in {".db", ".sqlite", ".sqlite3", ".key", ".pem", ".pfx"}
        if is_private and not p.is_relative_to(OUTPUT): violation("private_outside_access")
        mode, flags = args[1], args[2]
        writing = (isinstance(mode, str) and any(v in mode for v in "wax+")) or (
            type(flags) is int and bool(flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND)))
        if writing and not p.is_relative_to(OUTPUT): violation("outside_writes")
    if event in {"os.mkdir", "os.remove", "os.rmdir", "os.rename", "os.chmod", "os.link", "os.symlink"}:
        values = args[:2] if event in {"os.rename", "os.link", "os.symlink"} else args[:1]
        for value in values:
            if isinstance(value, (str, bytes, os.PathLike)) and not Path(os.fsdecode(value)).absolute().is_relative_to(OUTPUT):
                violation("outside_writes")


_sqlite_connect = sqlite3.connect


class AuditedConnection(sqlite3.Connection):
    def execute(self, sql, parameters=()):
        AUDIT["SQL_attempts"] += 1
        kind = str(sql).strip().split()[0].upper()
        AUDIT["SQL_kinds"][kind] = AUDIT["SQL_kinds"].get(kind, 0) + 1
        try:
            result = super().execute(sql, parameters)
            AUDIT["SQL_execute_returned"] += 1
            return result
        except BaseException:
            AUDIT["SQL_execute_failed"] += 1
            raise

    def rollback(self):
        AUDIT["SQLite_rollback_calls"] += 1
        return super().rollback()

    def set_authorizer(self, callback):
        def policy(action, first, second, database, trigger):
            if action in {sqlite3.SQLITE_ATTACH, sqlite3.SQLITE_DETACH} or (
                    action == sqlite3.SQLITE_FUNCTION and str(second).lower() == "load_extension"):
                AUDIT["SQL_authorizer_denials"] += 1
                return sqlite3.SQLITE_DENY
            return sqlite3.SQLITE_OK if callback is None else callback(action, first, second, database, trigger)
        return super().set_authorizer(policy)


def guarded_connect(database, *args, **kwargs):
    if database == ":memory:":
        maximum = 16 if PHASE == "new54" else 8 if PHASE == "r19_24" else 32
        if PHASE not in {"new54", "r19_24", "legacy_A1"} or AUDIT["memory_SQLite_connections"] >= maximum:
            violation("private_outside_access")
        AUDIT["memory_SQLite_connections"] += 1
    else:
        if PHASE != "new54" or RUN is None: violation("private_outside_access")
        raw = str(database)
        if raw.startswith("file:"):
            parsed = urlparse(raw)
            if parsed.netloc or parsed.query not in {"mode=rw", "mode=ro"}: violation("private_outside_access")
            raw = unquote(parsed.path)
            if re.match(r"^/[A-Za-z]:", raw): raw = raw[1:]
        target(raw)
        if not Path(raw).absolute().is_relative_to(RUN): violation("private_outside_access")
        AUDIT["file_SQLite_connections"] += 1
        if AUDIT["file_SQLite_connections"] > 104: violation("private_outside_access")
    key = CURRENT_ID or "setup"
    AUDIT["SQLite_by_case"][key] = AUDIT["SQLite_by_case"].get(key, 0) + 1
    if PHASE == "new54" and AUDIT["SQLite_by_case"][key] > 2: violation("private_outside_access")
    kwargs["factory"] = AuditedConnection
    connection = _sqlite_connect(database, *args, **kwargs)
    connection.set_authorizer(None)
    OWNED_CONNECTIONS.append(connection)
    return connection


def memory_connection():
    connection = sqlite3.connect(":memory:", isolation_level=None, check_same_thread=False)
    connection.row_factory = sqlite3.Row
    return connection


def preflight(suite):
    receipt = load(RECORD / "APPROVAL_RECEIPT.json")
    if receipt["approved"] is not True or sha(receipt["request_path"]) != receipt["request_sha256"]:
        raise RuntimeError("R21_APPROVAL_REQUIRED")
    expected = receipt["legacy_python"] if suite == "legacy44" else receipt["base_python"]
    if sys.executable != expected or sys.version_info[:3] != (3, 12, 14): raise RuntimeError("PYTHON_SCOPE_MISMATCH")
    baseline = load(RECORD / "sources/PRECHECK_LOCK.json")
    mods = {x["path"] for x in baseline["allowed_modifications"]}
    changed = []
    for row in baseline["protected"]:
        if row["path"] in mods:
            changed.append(row["path"])
        elif sha(row["path"]) != row["sha256"]:
            raise RuntimeError("PROTECTED_HISTORY_CHANGED")
    for row in baseline["allowed_modifications"]:
        if sha(RECORD / "sources/original" / (Path(row["path"]).name + ".txt")) != row["before_sha256"]:
            raise RuntimeError("ORIGINAL_SOURCE_LOST")
    filelock = load(RECORD / "R22_FILE_LOCK.json")
    for row in filelock["executed_code"]:
        if sha(row["path"]) != row["sha256"]: raise RuntimeError("SOURCE_CHANGED_SINCE_TEST_FREEZE")
        ast.parse(Path(row["path"]).read_text(encoding="utf-8"))
    for row in filelock["inputs"]:
        if sha(row["path"]) != row["sha256"]:raise RuntimeError("FIXTURE_CHANGED_SINCE_FREEZE")
    return baseline, filelock, changed


def module_at(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec); sys.modules[name] = module
    spec.loader.exec_module(module)
    return module


class Results(unittest.TextTestResult):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs); self.rows = []

    def startTest(self, test):
        global CURRENT_ID
        CURRENT_ID = test._testMethodName.split("_")[1]
        self.started = time.perf_counter_ns()
        super().startTest(test)

    def _exc_info_to_string(self, error, test):
        return redact(super()._exc_info_to_string(error, test))

    def row(self, test, status, detail=None):
        value = {"id": test._testMethodName.split("_")[1], "status": status,
                 "elapsed_ns": time.perf_counter_ns() - self.started}
        if detail: value["detail"] = redact(detail)
        self.rows.append(value)

    def addSuccess(self, test): super().addSuccess(test); self.row(test, "PASS")
    def addFailure(self, test, error):
        super().addFailure(test, error); self.row(test, "FAIL", self._exc_info_to_string(error, test))
    def addError(self, test, error):
        super().addError(test, error); self.row(test, "ERROR", self._exc_info_to_string(error, test))
    def addSkip(self, test, reason): super().addSkip(test, reason); self.row(test, "SKIP", reason)


def run_class(cls, ids, log):
    suite = unittest.defaultTestLoader.loadTestsFromTestCase(cls)
    actual = [test._testMethodName.split("_")[1] for test in suite]
    if set(actual) != set(ids) or len(actual)!=len(ids): raise RuntimeError("FROZEN_TEST_IDENTITIES_MISMATCH")
    by_id={t._testMethodName.split("_")[1]:t for t in suite}
    suite=unittest.TestSuite([by_id[i] for i in ids])
    return unittest.TextTestRunner(stream=log, verbosity=2, resultclass=Results).run(suite)


class ProjectedOutput:
    """Test-only logical R19 path; all physical I/O is mapped into this R21 run."""
    def __init__(self, logical, physical): self.logical, self.physical = logical, physical
    def __truediv__(self, name): return ProjectedOutput(self.logical / name, self.physical / name)
    def __str__(self): return str(self.logical)
    def write_bytes(self, value): return target(self.physical).write_bytes(value)


class ProjectedRead:
    def __init__(self, physical): self.physical = physical
    @property
    def parents(self): return self.physical.parents
    def exists(self): return self.physical.exists()
    def is_symlink(self): return self.physical.is_symlink()
    def lstat(self): return self.physical.lstat()
    def stat(self): return self.physical.stat()
    def read_bytes(self): return self.physical.read_bytes()


def install_R19_path_projection(run):
    from auth_lifecycle.online_state import dsat_authority_runtime as runtime
    logical = BASE / "2026-11-11_r19_offline" / ("r21_virtual_" + run.name)
    original = runtime.Path

        p = original(value)
        if p.is_relative_to(logical):

def worker(input_path):
    """Finite owned kernel-object worker; never a service or database process."""
    global PHASE, RUN
    PHASE = "worker"
    path = target(input_path)
    data = load(path); RUN = target(data["run_root"])
    if data.get("source_kind") != "R22_OWNED_STUB": raise RuntimeError("STUB_SCOPE_DENIED")
    sys.path.insert(0, str(PROJECT / "src"))
    from auth_lifecycle.online_state import dsat_windows_runtime as w
    scope = w.authorize_test_scope(RUN)
    root = scope.path(data["resource_root"])
    mode = data["mode"]
    lease = None; inherited = None; event = None; manager=None
    try:
        if mode == "handoff":
            inherited = int(data["inherited_handle"])
            facts = w._handle_file_facts(inherited)
            if Path(facts["final_path"]) != root / "instance.lock": raise RuntimeError("INHERITED_LOCK_MISMATCH")
            print(json.dumps({"status":"INHERITED_LOCK_OWNED","pid":os.getpid()}),flush=True)
            sys.stdin.buffer.readline(16)
        elif mode == "event":
            from auth_lifecycle.online_state import dsat_authority_lifecycle as lifecycle
            name = data["event_name"]
            if not re.fullmatch(r"Local\\auth_lifecycle_r22_[a-f0-9]{32,64}", name): raise RuntimeError("EVENT_SCOPE_DENIED")
            kernel,_=w._api();event=kernel.OpenEventW(0x100000,False,name)
            if not event: raise RuntimeError("EVENT_UNAVAILABLE")
            manager=lifecycle.supervise_persistent_offline(scope,root,data["authorization_id"],control_name=name)
            print(json.dumps({"status":"WAITING_EVENT","pid":os.getpid()}),flush=True)
            return 0 if kernel.WaitForSingleObject(event,5000)==0 else 2
        elif mode in {"race","crash"}:
            from auth_lifecycle.online_state import dsat_authority_lifecycle as lifecycle
            from auth_lifecycle.online_state import dsat_authority_live as live
            if mode=="crash":
                lease=w.reserve_instance(scope,root)
                instance=lifecycle.load_persistent_instance(scope,root)
                approval=live.load_start_approval(scope,root,instance,data["authorization_id"],lease)
                if int(data["exit_code"])==73:
                    live.consume_start_approval(approval,lease,approval.body()["process_epoch"])
                else:
                    live.write_marker(scope,root,instance,"RECOVERY_REQUIRED",authorization_id=data["authorization_id"])
                scope.write_new(root/"partial_receipt.json",json.dumps({"phase":"STOPPING","pid":os.getpid()}).encode())
                os._exit(int(data["exit_code"]))
            try:
                manager=lifecycle.supervise_persistent_offline(scope,root,data["authorization_id"],request_context={"client_alias":"race:"+str(os.getpid())})
                status="LOCKED"
            except w.SafetyError as error:
                if error.code not in {"SINGLE_ACTIVE_CONFLICT","REPLAY"}:raise
                status="CONFLICT"
            print(json.dumps({"status":status,"pid":os.getpid(),"durable_admission":manager is not None}),flush=True)
            if manager is not None:sys.stdin.buffer.readline(16)
        elif mode=="cross_first":
            return cross_first(scope,data)
        elif mode=="cross_replay":
            return cross_replay(scope,data)
        else:raise RuntimeError("STUB_MODE_DENIED")
        return 0
    finally:
        if manager is not None:manager.stop()
        if lease is not None:lease.close()
        if inherited is not None:w._api()[0].CloseHandle(inherited)
        if event is not None:w._api()[0].CloseHandle(event)


def _expected_refusal(function):
    from auth_lifecycle.online_state.dsat_windows_runtime import SafetyError
    try:
        manager=function()
        if hasattr(manager,'stop'):manager.stop()
        return 'ADMITTED'
    except SafetyError as error:return error.code
