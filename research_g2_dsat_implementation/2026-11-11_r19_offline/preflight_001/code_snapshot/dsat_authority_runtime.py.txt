class RuntimeRefusal(ValueError):
    def __init__(self, code):
        allowed = {"CONFIG_INVALID", "PATH_DENIED", "PARAMETERS_UNCONFIRMED", "SOURCE_MISMATCH",
            "DEPENDENCY_RUNTIME_GAP", "D1B_NOT_AUTHORIZED", "P1_NOT_AUTHORIZED", "P2_NOT_AUTHORIZED",
            "CONTEXT_REQUIRED", "CONTEXT_CLOSED", "CONTEXT_ALREADY_BOUND", "METADATA_INVALID",
            "TARGET_MISMATCH", "PORT_CONFLICT", "OFFLINE_ADAPTER_FAILED", "UNSUPPORTED_ARGUMENTS"}
        self.code = code if code in allowed else "OFFLINE_ADAPTER_FAILED"
        super().__init__(self.code)


def _require(value, code="CONFIG_INVALID"):
    if not value:
        raise RuntimeRefusal(code)


def _id(value):
    _require(type(value) is str and re.fullmatch(r"[A-Za-z0-9_.:-]{1,160}", value) is not None)
    return value


def _path(value):
    _require(type(value) is str and "\x00" not in value, "PATH_DENIED")
    _require(re.match(r"^[A-Za-z]:[\\/]", value) is not None and not value.startswith(("\\\\", "//")), "PATH_DENIED")
    _require(":" not in value[2:] and "%" not in value, "PATH_DENIED")
    _require(not any(p in {".", ".."} or p.endswith((".", " ")) for p in re.split(r"[\\/]", value)[1:]), "PATH_DENIED")
    return ntpath.normcase(ntpath.normpath(value))


def _approved(doc, name):

    return entry["approved_value"]

class RuntimeContext:
    __slots__ = ("__weakref__",)

    def __init__(self, *args, **kwargs):
        raise RuntimeRefusal("CONTEXT_REQUIRED")

    @classmethod
    def from_offline_host(cls, config, *, process, source_digest):
        _require(type(config) is RuntimeConfig and type(process) is dict and set(process) == {"pid", "created_at", "epoch", "python", "source_kind"})
        _require(type(process["pid"]) is int and 1 <= process["pid"] <= 2147483647
             and process["source_kind"] == "OFFLINE_SIMULATION" and _path(process["python"]) == _path(PYTHON))
        _id(process["created_at"]); _id(process["epoch"])
        _require(type(source_digest) is str and re.fullmatch(r"[0-9a-f]{64}", source_digest) is not None, "SOURCE_MISMATCH")
        context = object.__new__(cls)
        _CONTEXTS[context] = {"config": config, "process": dict(process), "source_digest": source_digest,
            "store": None, "connection": None, "lock": None, "server": None,
            "closed": False, "bound": False, "events": [], "registrations": 0}
        return context

    def _record(self):
        rec = _CONTEXTS.get(self)
        _require(rec is not None, "CONTEXT_REQUIRED")
        _require(not rec["closed"], "CONTEXT_CLOSED")
        return rec

    def validate_service_entry(self, role, root, ticket=None):
        self._record()
        # No live context issuer is provided in D1-A. Fail before config I/O.
        raise RuntimeRefusal("D1B_NOT_AUTHORIZED")

    def bind_store(self, store):
        rec = self._record()
        _require(not rec["bound"], "CONTEXT_ALREADY_BOUND")
        rec.update(store=store, connection=store.connection, lock=store.lock, bound=True)
        rec["registrations"] += 1
        rec["events"].append("dormant_bind")

    def _bound(self, store=None):
        rec = self._record()
        _require(rec["bound"] and (store is None or store is rec["store"]), "SOURCE_MISMATCH")
        _require(rec["store"].connection is rec["connection"] and rec["store"].lock is rec["lock"], "SOURCE_MISMATCH")
        return rec

    def on_listener_ready(self, server, store):
        rec = self._bound(store)
        _require(server.server_address == ("127.0.0.1", 27443) and server.TLS_REQUIRED is True, "PORT_CONFLICT")
        rec["server"] = server
        rec["events"].append("listener_simulated")

    def metadata(self):
        rec = self._bound()
        return {"schema": META_SCHEMA, "environment_id": ENVIRONMENT, "service_role": "authority",
            "source_kind": "OFFLINE_SIMULATION", "process": dict(rec["process"]),
            "source_digest": rec["source_digest"], "config_digest": rec["config"].config_digest,
            "endpoint": {"address": "127.0.0.1", "port": 27443, "TLS_REQUIRED": True},
            "references": {"DB": "authority/authority.db", "key": "secure/authority.key", "key_type": "NEW_TEST_STATE_KEY"},
            "mount_state": "DORMANT_SOURCE_BOUND", "P2_read_enabled": False, "permits_issued": 0,
            "session_reads": 0, "registrations": rec["registrations"], "missing_fields": list(GAPS),
            "INSTANCE_RUNNING": False}

    def issue_permit(self):
        self._record()
        raise RuntimeRefusal("P2_NOT_AUTHORIZED")

    read_state = issue_permit

    def close_owned(self):
        rec = _CONTEXTS.get(self)
        _require(rec is not None, "CONTEXT_REQUIRED")
        if rec["closed"]:
            return
        rec["closed"] = True
        failed = False
        try:
            if rec["server"] is not None:
                try:
                    rec["server"].shutdown()
                except Exception:
                    failed = True
                finally:
                    try:
                        rec["server"].server_close()
                    except Exception:
                        failed = True
        finally:
            if rec["connection"] is not None:
                try:
                    rec["connection"].close()
                except Exception:
                    failed = True
        if failed:
            raise RuntimeRefusal("OFFLINE_ADAPTER_FAILED") from None

    def counters(self):
        rec = _CONTEXTS.get(self)
        _require(rec is not None, "CONTEXT_REQUIRED")
        return {"registrations": rec["registrations"], "closed": rec["closed"],
                "permits": 0, "session_reads": 0, "events": list(rec["events"])}

    def __repr__(self):
        return "<AuthorityRuntimeContext redacted>"

    def __reduce_ex__(self, protocol):
        raise TypeError("RUNTIME_CONTEXT_NOT_SERIALIZABLE")


def validate_runtime_context(context, role, root, ticket=None):
    _require(type(context) is RuntimeContext, "CONTEXT_REQUIRED")
    return context.validate_service_entry(role, root, ticket)


def simulate_bootstrap(config, context, adapters):
    _require(type(config) is RuntimeConfig and type(context) is RuntimeContext, "CONTEXT_REQUIRED")
    rec = context._record()
    _require(rec["config"] == config, "SOURCE_MISMATCH")
    validate_path_facts(config, adapters.path_facts(config))
    check_dependency_facts(adapters.dependency_facts())
    rec["events"].extend(["config_preflight", "dependency_preflight"])
    try:
        store = adapters.construct_store(config)
        rec["events"].append("store_constructed")
        context.bind_store(store)
        try:
            server = adapters.prepare_listener("127.0.0.1", 27443, TLS_REQUIRED=True)
        except Exception:
            raise RuntimeRefusal("PORT_CONFLICT") from None
        context.on_listener_ready(server, store)
        return context.metadata()
    except RuntimeRefusal:
        raise
    except Exception:
        raise RuntimeRefusal("OFFLINE_ADAPTER_FAILED") from None
    finally:
        context.close_owned()


def validate_metadata(value):
    fields = {"schema", "environment_id", "service_role", "source_kind", "process", "source_digest", "config_digest",
        "endpoint", "references", "mount_state", "P2_read_enabled", "permits_issued", "session_reads",
        "registrations", "missing_fields", "INSTANCE_RUNNING"}
    _require(type(value) is dict and set(value) == fields and value["schema"] == META_SCHEMA, "METADATA_INVALID")
    _require(value["environment_id"] == ENVIRONMENT and value["service_role"] == "authority"
        and value["source_kind"] == "OFFLINE_SIMULATION" and value["P2_read_enabled"] is False
        and value["INSTANCE_RUNNING"] is False and value["permits_issued"] == 0 and value["session_reads"] == 0
        and value["registrations"] == 1 and value["mount_state"] == "DORMANT_SOURCE_BOUND", "METADATA_INVALID")
    _require(all(type(value[k]) is int for k in ("permits_issued", "session_reads", "registrations")), "METADATA_INVALID")
    _require(value["endpoint"] == {"address": "127.0.0.1", "port": 27443, "TLS_REQUIRED": True}, "METADATA_INVALID")
    _require(value["references"] == {"DB": "authority/authority.db", "key": "secure/authority.key", "key_type": "NEW_TEST_STATE_KEY"}, "METADATA_INVALID")
    p = value["process"]
    _require(type(p) is dict and set(p) == {"pid", "created_at", "epoch", "python", "source_kind"}
        and type(p["pid"]) is int and p["pid"] > 0 and p["source_kind"] == "OFFLINE_SIMULATION"
        and _path(p["python"]) == _path(PYTHON), "METADATA_INVALID")
    _id(p["created_at"]); _id(p["epoch"])
    _require(value["missing_fields"] == list(GAPS), "METADATA_INVALID")
    _require(all(type(value[k]) is str and re.fullmatch(r"[0-9a-f]{64}", value[k]) for k in ("source_digest", "config_digest")), "METADATA_INVALID")
    _require(len(canonical(value)) <= 16384, "METADATA_INVALID")
    return json.loads(canonical(value))


def inspect_offline_metadata(request, provider):
    _require(type(request) is dict and set(request) == {"environment_id", "pid", "created_at", "epoch", "source_digest", "config_digest"}, "TARGET_MISMATCH")
    _require(request["environment_id"] == ENVIRONMENT and type(request["pid"]) is int, "TARGET_MISMATCH")
    try:
        observed = provider.read_one(request["pid"])
    except Exception:
        raise RuntimeRefusal("METADATA_INVALID") from None
    value = validate_metadata(observed)
    p = value["process"]
    _require(all(request[k] == p[k] for k in ("pid", "created_at", "epoch"))
        and all(request[k] == value[k] for k in ("source_digest", "config_digest")), "TARGET_MISMATCH")
    return value
