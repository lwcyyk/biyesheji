class LockSpy:
    def __init__(self):
        self._lock = threading.RLock()
        self.depth = 0
        self.entries = 0
        self.max_depth = 0

    def __enter__(self):
        self._lock.acquire()
        self.depth += 1; self.entries += 1
        self.max_depth = max(self.max_depth, self.depth)
        return self

    def __exit__(self, *args):
        self.depth -= 1; self._lock.release()


class ReaderSpy:
    def __init__(self, state, version=120, operation_limit=None):
        self.state = state
        self.version = version
        self.lock = LockSpy()
        self.calls = 0
        self.operations = 0
        self.operation_limit = operation_limit
        self.forbidden_calls = 0
        self.fail_message = None

    def read_state(self):
        if self.lock.depth != 1:
            raise AssertionError("SOURCE_LOCK_REQUIRED")
        self.calls += 1; self.operations += 1
        if self.fail_message is not None:
            raise RuntimeError(self.fail_message)
        return self.version, self.state

    def call(self, *args):
        self.forbidden_calls += 1
        raise AssertionError("OLD_DISPATCH_FORBIDDEN")

    def execute(self, *args):
        self.forbidden_calls += 1
        raise AssertionError("SQL_FORBIDDEN")

    def __repr__(self):
        return "<SyntheticReaderSpy redacted>"


class B1Cases(unittest.TestCase):
    def setUp(self):
        self.case_id = self._testMethodName.split("_")[1]
        self.spec = next(x for x in CONFIG["plan"]["cases"] if x["id"] == self.case_id)
        self.state = json.loads((CONFIG["fixture_root"]/"BASE_STATE.json").read_text(encoding="utf-8"))
        self.meta = json.loads((CONFIG["fixture_root"]/"METADATA.json").read_text(encoding="utf-8"))

        self.checks = []
        self.started = time.perf_counter_ns()

    def tearDown(self):
        DETAILS.append({"id": self.case_id, "name": self.spec["name"], "checks": self.checks,
                        "calls": self.observed, "elapsed_ns": time.perf_counter_ns()-self.started})

    def check(self, condition, label):
        self.checks.append({"label": label, "satisfied": bool(condition)})
        if not condition:
            raise AssertionError(label)  # Fixed label only; no raw fixture repr in failures.

    def context(self, reader=None, scope=None, **overrides):
        data = dict(self.meta["host_context"]); data.update(overrides)
        return TrustedReadContext.from_host(reader or self.reader, scope or self.meta["scope_A"], **data)

    def read(self, *, reader=None, scope=None, context=None, limits=ReadLimits(), foreign=True):
        source = reader or self.reader
        ctx = self.context(source) if context is None else context
        requested = self.meta["scope_A"] if scope is None else scope
        before = canonical(source.state)
        version, calls = source.version, source.calls
        stdout, stderr, log = io.StringIO(), io.StringIO(), io.StringIO()
        handler = logging.StreamHandler(log); logger = logging.getLogger(); logger.addHandler(handler)
        started = time.perf_counter_ns()
        try:
            with contextlib.redirect_stdout(stdout), contextlib.redirect_stderr(stderr):
                result = read_dsat_authority_snapshot(source, requested, ctx, limits)
        finally:
            logger.removeHandler(handler)
        elapsed = time.perf_counter_ns()-started
        public = result.to_dict()
        self.check(before == canonical(source.state) and version == source.version, "AUTHORITY_WORLD_UNCHANGED")
        self.check(source.forbidden_calls == 0, "NO_OLD_DISPATCH_SQL_OR_WRITE")
        self.check(source.calls-calls <= 1, "AT_MOST_ONE_SOURCE_READ")
        self.check(public["global_complete"] is False and public["math_ready"] is False
                   and public["current_native_eligible"] is False, "NO_STRONG_QUALIFICATION")
        self.check(public["obligation_revision"] is None and public["authority_write_action"] == "NONE", "NO_O_REVISION_OR_COMMIT")
        rendered = result.to_json().decode()+repr(result)+json.dumps(public)+stdout.getvalue()+stderr.getvalue()+log.getvalue()
        self.no_secrets(rendered)
        if foreign:
            self.no_foreign(rendered)
        self.observed.append({"result": public, "total_ns": elapsed, "world_unchanged": True,
                              "source_reads": source.calls-calls, "log_bytes": len(log.getvalue()),
                              "stdout_bytes": len(stdout.getvalue()), "stderr_bytes": len(stderr.getvalue()),
                              "secret_scan": "CLEAN", "scope_scan": "CLEAN" if foreign else "EXPLICIT_WIDE_SCOPE"})
        return result

    def no_secrets(self, text):
        self.check(all(value not in text for value in self.canaries["sensitive_values"]), "SECRET_CANARIES_ABSENT")

    def no_foreign(self, text):
        self.check(all(value not in text for value in self.canaries["foreign_scope_A"]), "FOREIGN_SCOPE_METADATA_ABSENT")

    def ready(self, result):
        self.check(result.to_dict()["status"] == "STAGING_READY_WITH_GAPS", "STAGING_READY_WITH_EXPLICIT_GAPS")

    def request(self, public, rid):
        return next(x for x in public["requests"] if x["rid"] == rid)

    def resign(self, state, rid):
        raw = state["requests"][rid]
        raw["digest"] = digest({"caller": raw["origin"]["caller"], "request": raw["request"]})
