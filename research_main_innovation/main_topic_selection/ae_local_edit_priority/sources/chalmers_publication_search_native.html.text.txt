    function createSourceText(publication) {
        let ret = '';
        if (publication.Source != null) {
            if (publication.Source.SourceSerial != null) {
                ret += publication.Source.SourceSerial.Title || '';
            } else {
                ret += publication.Source.Title || '';
            }
            if (publication.Source.Volume != null && publication.Source.Volume.length > 0) {
                ret += (ret != '' ? '. ' : '') + 'Vol. ' + publication.Source.Volume;
            }
            if (publication.Source.Issue != null && publication.Source.Issue.length > 0) {
                ret += (ret != '' ? ' ' : '') + '(' + publication.Source.Issue + ')';
            }
            if (publication.Source.PageStart != null && publication.Source.PageStart.length > 0) {
                ret += (ret != '' ? ', ' : '') + 'p. ' + publication.Source.PageStart + '-' + publication.Source.PageEnd;
            }
        }
        return ret;
    }

    function createOaUrl(publication) {
        let ret = '';
        if (publication.DataObjects != null) {
            for (var i = 0; i < publication.DataObjects.length; i++) {
                let pub = publication.DataObjects[i];
                let isEmbargoDateNullOrValid = pub.EmbargoDate == null || (Date.parse(pub.EmbargoDate) < Date.now());

                if (pub.IsOpenAccess && pub.IsMainFulltext && isEmbargoDateNullOrValid && pub.Url != null) {
                    ret += pub.Url;
                    if (pub.Url.indexOf('http') < 0) {
                        ret = baseUrl + "/" + ret;
                    }
                    break;
                }
            }
        }
        return ret;
    }

                pubtype = "book";
            }
            else if (publication.PublicationType.NameEng.toLowerCase().indexOf("preprint") != -1) {
                pubtype = "preprint";
            }
            else {
                pubtype = "article";
            }
        }

        if (pubtype == "book") {
            ret = ('url_ver=Z39.88-2004&rft_val_fmt=info%3Aofi%2Ffmt%3Akev%3Amtx%3Abook&rft.genre=' + pubtype + '&rft.date=' + year + doi + '&rft.btitle=' + title + authors + '&rft.isbn=' + isbn + publisher).replace(/ /g,"+");
        }
        else if (pubtype == "dissertation") {
            ret = ('url_ver=Z39.88-2004&rft_val_fmt=info%3Aofi%2Ffmt%3Akev%3Amtx%3Abook&rft.genre=' + pubtype + '&rft.date=' + year + doi + '&rft.btitle=' + title + authors + '&rft.isbn=' + isbn + publisher).replace(/ /g,"+");
        }
        else if (pubtype == "bookitem")
        {
            ret = ("url_ver=Z39.88-2004&rft_val_fmt=info%3Aofi%2Ffmt%3Akev%3Amtx%3Abook&rft.genre=" + pubtype + '&rft.date=' + year + doi + "&rft.btitle=" + jtitle + "&rft.atitle=" + title + volume + authors + issue + pagestart + pageend + issn + eissn + '&rft.isbn=' + isbn + publisher).Replace(" ", "+");
        }
        else {
            ret = ('url_ver=Z39.88-2004&rft_val_fmt=info%3Aofi%2Ffmt%3Akev%3Amtx%3Ajournal&rft.genre=' + pubtype + '&rft.date=' + year + doi + '&rft.jtitle=' + jtitle + '&rft.atitle=' + title + volume + authors + issue + pagestart + pageend + issn + eissn).replace(/ /g,"+");
        }

        return ret;
    }

    function updatePublications(resetToFirstPage) {
        viewModel.isUpdatingPublicationResults(viewModel.isUpdatingPublicationResults() + 1);

        if (resetToFirstPage) {
            viewModel.publicationDisplayStart(0);
            viewModel.publicationCurrentPage(0);
        }

        updatePublicationsSearchAndFilterQueryParams();

        $.get("/umbraco/surface/publicationqueriessurface/GetPublicationsForOrganizationWithStandardAggregations?" + viewModel.publicationsSearchAndFilterQueryParams())
            .done(function (res) {
                populateWithNewData(res);
                viewModel.isUpdatingPublicationResults(viewModel.isUpdatingPublicationResults() - 1);
            })
            .fail(function (jqXHR, textStatus, errorThrown) {
                console.error('ERROR: ' + textStatus);
                viewModel.isUpdatingPublicationResults(viewModel.isUpdatingPublicationResults() - 1);
            });
    }

    function populateWithNewData(data) {
        viewModel.departmentFilters(data.Departments);
        viewModel.otherOrganizationFilters(data.OtherOrganizations);
        viewModel.publicationTypeFilters(data.PublicationTypesAggregation);
        viewModel.yearFilters(data.YearsAggregation);
        viewModel.publicationsTotalCount(data.PublicationsCount);
        viewModel.publications(_.map(data.PublicationRawHits, '_source'));

        updatePagination();
    }

    function updatePagination() {
        let publicationsPerPage = viewModel.publicationsPerPage; // Hardcoded in backend
        let publicationsCount = viewModel.publicationsTotalCount();
        let pageIndexes = [0];
        let pageCount = 1;
        let currentPage = 0;
        if (publicationsCount > publicationsPerPage) {
            pageCount = Math.ceil(publicationsCount / publicationsPerPage);
            currentPage = Math.floor(viewModel.publicationDisplayStart() / publicationsPerPage);

            pageIndexes = [];
            var slidingOffset = 0;
            if (currentPage == 1) {
                slidingOffset = 1
            } else if (currentPage > 1) {
                slidingOffset = 2;
            }

            var maxPageIndex = -1;
            for (var i=0; i<(pageCount < publicationsPerPage ? pageCount : publicationsPerPage); i++) {
                var pageIndex = currentPage + i - slidingOffset;
                if (pageIndex < pageCount) {
                    pageIndexes.push(pageIndex);
                } else {
                    if (maxPageIndex === -1) {
                        maxPageIndex = pageIndex - 1;
                    }
                    var complementPageIndex = maxPageIndex - pageIndexes.length;
                    if (complementPageIndex >= 0) {
                        pageIndexes.push(complementPageIndex);
                    }
                }
            }
            pageIndexes.sort(function (a,b) {
                return a > b;
            });
        }

        viewModel.publicationPageCount(pageCount);
        viewModel.publicationCurrentPage(currentPage);
        viewModel.publicationPageIndexes(pageIndexes);
    }

    function stepBackOnePage() {
        viewModel.publicationDisplayStart(viewModel.publicationDisplayStart() - 1 * viewModel.publicationsPerPage);
        updatePublications();
    }

    function stepForwardOnePage() {
        viewModel.publicationDisplayStart(viewModel.publicationDisplayStart() + 1 * viewModel.publicationsPerPage);
        updatePublications();
    }

    function stepToPage(pageIndex) {
        viewModel.publicationDisplayStart(pageIndex * viewModel.publicationsPerPage);
        updatePublications();
    }
